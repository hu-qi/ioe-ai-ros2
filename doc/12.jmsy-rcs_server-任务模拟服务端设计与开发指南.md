### 1.2 RCS-2000 V3.3 接口映射

| 客户端调用 | 服务端接口 | 说明 |
|:---|:---|:---|
| `genAgvSchedulingTask` | `POST /genAgvSchedulingTask` | 下发搬运任务 |
| `cancelTask` | `POST /cancelTask` | 取消任务 |
| `continueTask` | `POST /continueTask` | 继续暂停任务 |
| `queryAgvStatus` | `POST /queryAgvStatus` | 查询 AGV 状态 |
| `queryTaskStatus` | `POST /queryTaskStatus` | 查询任务状态 |
| `queryPodBerthAndMat` | `POST /queryPodBerthAndMat` | 查询仓位绑定关系 |
| `bindCtnrAndBin` | `POST /bindCtnrAndBin` | 绑定/解绑容器与仓位 |
| 服务端回调 | `POST {callback_url}/agvCallback` | 任务状态变更通知 |
| 服务端回调 | `POST {callback_url}/warnCallback` | 告警通知 |

### 1.3 响应格式统一标准

```json
{
    "code": "0",           // "0"=成功, "-1"=失败
    "msg": "操作成功",     // 描述消息
    "data": {}             // 业务数据
}
```

#!/usr/bin/env python3
"""
jmsy-rcs_server — RCS-2000 V3.3 任务工作流模拟服务
IP: 192.168.31.191:8080

模拟完整任务生命周期：接收任务 → 自动执行 → 状态推进 → 回调通知客户端
"""

import logging
import threading
import time
import json
from typing import Dict, Any, List, Optional

import uvicorn
from fastapi import FastAPI, HTTPException, Request
from fastapi.responses import JSONResponse

from bay_manager import BayManager
from agv_manager import AGVManager
from task_manager import TaskManager
from callback_sender import CallbackSender

# 日志
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s [RCS-MOCK] %(levelname)s %(message)s'
)
logger = logging.getLogger("rcs_mock")


class RCSMockServer:
    """RCS-2000 V3.3 模拟服务"""

    def __init__(self, callback_url: str = "http://127.0.0.1:8080/eyeSky/robot/reporter"):

        # 子模块
        self.bay_mgr = BayManager()
        self.agv_mgr = AGVManager()
        self.callback = CallbackSender(callback_url, logger)
        self.task_mgr = TaskManager(
            self.bay_mgr, self.agv_mgr, self.callback, logger
        )

        # FastAPI
        self.app = FastAPI(
            title="RCS-2000 Mock Server",
            version="3.3",
            description="任务工作流模拟服务"
        )
        self._setup_routes()

    # ═══════════════════════════════════════════════════
    # 路由注册
    # ═══════════════════════════════════════════════════

    def _setup_routes(self):

        @self.app.get("/")
        async def root():
            return {"service": "RCS-2000 Mock Server", "version": "3.3"}

        @self.app.get("/status")
        async def status():
            return {
                "agvs": self.agv_mgr.get_all(),
                "bays": self.bay_mgr.summary(),
                "tasks": self.task_mgr.summary(),
            }

        # ── 任务接口 ──

        @self.app.post("/genAgvSchedulingTask")
        async def gen_task(request: Request):
            """2.1.1 生产任务单"""
            body = await request.json()
            logger.info(f"[genAgvSchedulingTask] 请求: {json.dumps(body, ensure_ascii=False)}")
            result = self.task_mgr.create_task(body)
            return JSONResponse(content=result)

        @self.app.post("/cancelTask")
        async def cancel_task(request: Request):
            """2.1.3 取消任务"""
            body = await request.json()
            logger.info(f"[cancelTask] 请求: taskCode={body.get('taskCode')}")
            result = self.task_mgr.cancel_task(body)
            return JSONResponse(content=result)

        @self.app.post("/continueTask")
        async def continue_task(request: Request):
            """2.1.2 继续任务"""
            body = await request.json()
            logger.info(f"[continueTask] 请求: taskCode={body.get('taskCode')}")
            result = self.task_mgr.continue_task(body)
            return JSONResponse(content=result)

        # ── 查询接口 ──

        @self.app.post("/queryAgvStatus")
        async def query_agv(request: Request):
            """3.1.10 查询 AGV 状态"""
            body = await request.json()
            robot_ids = body.get('robotIDs', [])
            result = self.agv_mgr.query(robot_ids)
            return JSONResponse(content=result)

        @self.app.post("/queryTaskStatus")
        async def query_task(request: Request):
            """3.1.9 查询任务状态"""
            body = await request.json()
            task_codes = body.get('taskCodes', [])
            result = self.task_mgr.query_task_status(task_codes)
            return JSONResponse(content=result)

        @self.app.post("/queryPodBerthAndMat")
        async def query_bay(request: Request):
            """3.1.6 查询仓位绑定关系"""
            body = await request.json()
            bay_ids = body.get('stgBinCodes', [])
            result = self.bay_mgr.query(bay_ids)
            return JSONResponse(content=result)

        # ── 绑定接口 ──

        @self.app.post("/bindCtnrAndBin")
        async def bind_ctnr(request: Request):
            """3.1.8 容器与仓位绑定"""
            body = await request.json()
            logger.info(f"[bindCtnrAndBin] 请求: {json.dumps(body, ensure_ascii=False)}")
            result = self.bay_mgr.bind(**body)
            return JSONResponse(content=result)

        # ── 调试/管理接口 ──

        @self.app.post("/admin/reset")
        async def reset(request: Request):
            """重置所有状态"""
            self.bay_mgr.reset()
            self.agv_mgr.reset()
            self.task_mgr.reset()
            return {"code": "0", "msg": "已重置"}

        @self.app.post("/admin/scenario/{scenario_id}")
        async def apply_scenario(scenario_id: str):
            """应用预设场景"""
            from scenarios import apply_scenario
            result = apply_scenario(scenario_id, self)
            return result


def create_app(callback_url: str = "http://127.0.0.1:8080/eyeSky/robot/reporter"):
    server = RCSMockServer(callback_url=callback_url)
    return server.app


if __name__ == "__main__":
    import sys
    cb_url = sys.argv[1] if len(sys.argv) > 1 else "http://127.0.0.1:8080/eyeSky/robot/reporter"
    app = create_app(cb_url)
    logger.info(f"RCS Mock Server 启动在 192.168.31.191:8080, callback → {cb_url}")

### 3.2 `bay_manager.py` — 仓位管理器

```python
#!/usr/bin/env python3
"""
仓位管理器 — 24 起始仓位 + 18 终点仓位，管理绑定/解绑状态
"""

import threading
import time
from typing import Dict, List, Any


class BayManager:
    """仓位状态管理"""

    SOURCE_BAYS = [
        "A1001","A1002","A1003","A1004","A1005","A1006",
        "B1001","B1002","B1003","B1004","B1005","B1006",
        "C1001","C1002","C1003","C1004","C1005","C1006",
        "D1001","D1002","D1003","D1004","D1005","D1006",
    ]

    DEST_BAYS = [
        "T2001","T2002","T2003","T2004","T2005","T2006",
        "T3001","T3002","T3003","T3004","T3005","T3006",
        "T4001","T4002","T4003","T4004","T4005","T4006",
    ]

    def __init__(self):
        self._lock = threading.RLock()
        self._bays: Dict[str, dict] = {}
        self.reset()

    def reset(self):
        with self._lock:
            self._bays = {}
            # 初始化起始仓位：默认未绑定（无货）
            for bay_id in self.SOURCE_BAYS:
                self._bays[bay_id] = {
                    "stgBinCode": bay_id,
                    "ctnrCode": "",           # 空=无货
                    "ctnrTyp": "0",
                    "indBind": "0",           # 0=未绑定
                    "bindTime": "",
                }
            # 初始化终点仓位：默认空
            for bay_id in self.DEST_BAYS:
                self._bays[bay_id] = {
                    "stgBinCode": bay_id,
                    "ctnrCode": "",
                    "ctnrTyp": "0",
                    "indBind": "0",
                    "bindTime": "",
                }

    # ── 绑定/解绑 ──

    def bind(self, ctnrCode: str, ctnrTyp: str, stgBinCode: str,
             indBind: str) -> Dict[str, Any]:
        """容器与仓位绑定/解绑"""
        with self._lock:
            bay = self._bays.get(stgBinCode)
            if not bay:
                return {"code": "-1", "msg": f"仓位不存在: {stgBinCode}"}

            if indBind == "1":
                bay["ctnrCode"] = ctnrCode
                bay["ctnrTyp"] = ctnrTyp
                bay["indBind"] = "1"
                bay["bindTime"] = time.strftime('%Y-%m-%d %H:%M:%S')
            else:
                bay["ctnrCode"] = ""
                bay["ctnrTyp"] = "0"
                bay["indBind"] = "0"
                bay["bindTime"] = ""

            return {"code": "0", "msg": "绑定成功" if indBind == "1" else "解绑成功"}

    # ── 查询 ──

    def query(self, bay_ids: List[str]) -> Dict[str, Any]:
        """查询仓位绑定关系"""
        with self._lock:
            data = []
            for bay_id in bay_ids:
                bay = self._bays.get(bay_id)
                if bay:
                    data.append(dict(bay))
            return {"code": "0", "msg": "ok", "data": data}

    # ── 任务完成后的仓位状态变更 ──

    def on_task_arrival(self, dst_bay: str, src_bay: str):
        """任务到达终点仓位后：终点仓位变为有货(绑定)，起始仓位解绑"""
        with self._lock:
            # 终点仓位 → 有货（绑定）
            if dst_bay in self._bays:
                self._bays[dst_bay]["ctnrCode"] = f"CTNR_{dst_bay}"
                self._bays[dst_bay]["ctnrTyp"] = "1"
                self._bays[dst_bay]["indBind"] = "1"
                self._bays[dst_bay]["bindTime"] = time.strftime('%Y-%m-%d %H:%M:%S')

            # 起始仓位 → 解绑（货物已搬走）
            if src_bay in self._bays:
                self._bays[src_bay]["ctnrCode"] = ""
                self._bays[src_bay]["ctnrTyp"] = "0"
                self._bays[src_bay]["indBind"] = "0"

    def set_bound_simulated(self, bay_id: str, cargo_type: int = 1):
        """模拟仓位已绑定（用于测试场景预设）"""
        with self._lock:
            if bay_id in self._bays:
                self._bays[bay_id]["ctnrCode"] = f"CTNR_{bay_id}"
                self._bays[bay_id]["ctnrTyp"] = str(cargo_type)
                self._bays[bay_id]["indBind"] = "1"
                self._bays[bay_id]["bindTime"] = time.strftime('%Y-%m-%d %H:%M:%S')

    def summary(self) -> dict:
        with self._lock:
            bound_src = sum(1 for k in self.SOURCE_BAYS
                           if self._bays.get(k, {}).get("indBind") == "1")
            empty_dst = sum(1 for k in self.DEST_BAYS
                           if self._bays.get(k, {}).get("indBind") == "0")
            return {
                "source_total": len(self.SOURCE_BAYS),
                "source_bound": bound_src,
                "dest_total": len(self.DEST_BAYS),
                "dest_empty": empty_dst,
            }
```

### 3.3 `agv_manager.py` — AGV 管理器

```python
#!/usr/bin/env python3
"""
AGV 管理器 — 5 台 AGV 状态机（IDLE / BUSY / OFFLINE / ERROR）
"""

import threading
import time
from typing import Dict, List, Any


class AGVManager:
    """AGV 状态管理"""

    AGV_IDS = [1001, 1002, 1003, 1004, 1005]

    def __init__(self):
        self._lock = threading.RLock()
        self._agvs: Dict[int, dict] = {}
        self.reset()

    def reset(self):
        with self._lock:
            self._agvs = {}
            for agv_id in self.AGV_IDS:
                self._agvs[agv_id] = {
                    "robotID": agv_id,
                    "agvCode": str(agv_id),
                    "agvStatus": "IDLE",         # IDLE/BUSY/OFFLINE/ERROR
                    "battery": 85,
                    "positionCode": "",
                    "currentTaskCode": "",
                    "updateTime": time.strftime('%Y-%m-%d %H:%M:%S'),
                }

    # ── 状态更新 ──

    def set_busy(self, agv_id: int, task_code: str = ""):
        """标记 AGV 为忙碌"""
        with self._lock:
            agv = self._agvs.get(agv_id)
            if agv:
                agv["agvStatus"] = "BUSY"
                agv["currentTaskCode"] = task_code
                agv["updateTime"] = time.strftime('%Y-%m-%d %H:%M:%S')

    def set_idle(self, agv_id: int):
        """标记 AGV 为空闲"""
        with self._lock:
            agv = self._agvs.get(agv_id)
            if agv:
                agv["agvStatus"] = "IDLE"
                agv["currentTaskCode"] = ""
                agv["updateTime"] = time.strftime('%Y-%m-%d %H:%M:%S')

    def set_error(self, agv_id: int):
        """标记 AGV 为故障"""
        with self._lock:
            agv = self._agvs.get(agv_id)
            if agv:
                agv["agvStatus"] = "ERROR"
                agv["updateTime"] = time.strftime('%Y-%m-%d %H:%M:%S')

    def set_offline(self, agv_id: int):
        """标记 AGV 为离线"""
        with self._lock:
            agv = self._agvs.get(agv_id)
            if agv:
                agv["agvStatus"] = "OFFLINE"

    # ── 查询 ──

    def query(self, robot_ids: List[int]) -> Dict[str, Any]:
        """查询 AGV 状态"""
        with self._lock:
            data = []
            for rid in robot_ids:
                agv = self._agvs.get(rid)
                if agv:
                    data.append(dict(agv))
            return {"code": "0", "msg": "ok", "data": data}

    def get(self, agv_id: int) -> dict:
        return self._agvs.get(agv_id, {})

    def get_all(self) -> Dict[int, dict]:
        with self._lock:
            return {k: dict(v) for k, v in self._agvs.items()}

    def get_idle(self) -> int:
        """获取空闲 AGV 数量"""
        with self._lock:
            return sum(1 for a in self._agvs.values() if a["agvStatus"] == "IDLE")
```

### 3.4 `task_manager.py` — 任务管理器

```python
#!/usr/bin/env python3
"""
任务管理器 — 任务生命周期控制 + 自动状态推进 + 回调通知
"""

import threading
import time
import logging
from typing import Dict, Any


class TaskManager:
    """任务生命周期管理"""

    def __init__(self, bay_mgr, agv_mgr, callback, logger: logging.Logger):
        self.bay_mgr = bay_mgr
        self.agv_mgr = agv_mgr
        self.callback = callback
        self.logger = logger

        self._lock = threading.RLock()
        self._tasks: Dict[str, dict] = {}
        self._task_counter = 0
        self._timers: Dict[str, threading.Timer] = {}

    def reset(self):
        with self._lock:
            for timer in self._timers.values():
                timer.cancel()
            self._tasks = {}
            self._timers = {}
            self._task_counter = 0

    # ═══════════════════════════════════════════════════
    # 创建任务
    # ═══════════════════════════════════════════════════

    def create_task(self, body: dict) -> dict:
        """
        创建 AGV 调度任务

        请求体:
            taskCode, taskTyp, priority, agvCode, positionCodePath: [{positionCode, type}]

        流程:
            接收 → 校验 → 创建任务记录 → 异步执行 → 回调通知
        """
        task_code = body.get("taskCode", "")
        agv_code = body.get("agvCode", "")
        path = body.get("positionCodePath", [])
        src_bay = path[0]["positionCode"] if len(path) > 0 else ""
        dst_bay = path[1]["positionCode"] if len(path) > 1 else ""

        # 校验
        agv_id = int(agv_code) if agv_code else 0
        agv = self.agv_mgr.get(agv_id)
        if not agv:
            return {"code": "-1", "msg": f"AGV 不存在: {agv_code}"}

        if agv["agvStatus"] == "ERROR":
            return {"code": "-1", "msg": f"AGV 故障: {agv_code}"}

        # 生成 RCS 任务流水号
        self._task_counter += 1
        rcs_task_no = f"RCS_{int(time.time())}_{self._task_counter:04d}"

        # 标记 AGV 忙碌
        self.agv_mgr.set_busy(agv_id, rcs_task_no)

        # 记录任务
        with self._lock:
            self._tasks[rcs_task_no] = {
                "rcsTaskNo": rcs_task_no,
                "clientTaskCode": task_code,
                "agvCode": agv_code,
                "agvId": agv_id,
                "srcBay": src_bay,
                "dstBay": dst_bay,
                "taskTyp": body.get("taskTyp", "F13"),
                "priority": body.get("priority", 64),
                "status": "RUNNING",          # RUNNING / COMPLETED / FAILED / CANCELLED
                "createTime": time.strftime('%Y-%m-%d %H:%M:%S'),
                "step": 0,                    # 当前执行步骤
            }

        # 异步执行任务
        task = self._tasks[rcs_task_no]
        timer = threading.Timer(2.0, self._task_worker, args=[rcs_task_no, task])
        self._timers[rcs_task_no] = timer
        timer.start()

        self.logger.info(f"任务创建: {rcs_task_no} AGV={agv_code} "
                         f"src={src_bay} → dst={dst_bay}")

        return {
            "code": "0",
            "msg": "任务创建成功",
            "data": {"taskNo": rcs_task_no}
        }

    def _task_worker(self, rcs_task_no: str, task: dict):
        """任务执行Worker — 分步推进并回调客户端"""
        try:
            agv_id = task["agvId"]
            client_task_code = task["clientTaskCode"]

            # Step 1: 开始执行（延迟 1s）
            time.sleep(1.0)
            self.logger.info(f"[{rcs_task_no}] Step1: 开始执行")
            self.callback.send_agv_callback(
                taskCode=client_task_code,
                agvCode=str(agv_id),
                taskStatus="2",           # 执行中
                positionCode=task["srcBay"],
            )

            # Step 2: 模拟搬运过程（延迟 3s）
            time.sleep(3.0)
            self.logger.info(f"[{rcs_task_no}] Step2: 执行中...")
            self.callback.send_agv_callback(
                taskCode=client_task_code,
                agvCode=str(agv_id),
                taskStatus="2",           # 执行中
                positionCode="MOVING",
            )

            # Step 3: 到达目的仓位（延迟 2s）
            time.sleep(2.0)
            task["status"] = "COMPLETED"
            self.logger.info(f"[{rcs_task_no}] Step3: 到达终点 {task['dstBay']}")

            # 更新仓位状态
            self.bay_mgr.on_task_arrival(task["dstBay"], task["srcBay"])

            # 释放 AGV
            self.agv_mgr.set_idle(agv_id)

            # 回调完成通知
            self.callback.send_agv_callback(
                taskCode=client_task_code,
                agvCode=str(agv_id),
                taskStatus="9",           # 已完成
                positionCode=task["dstBay"],
            )

            self.logger.info(f"[{rcs_task_no}] 任务完成")

        except Exception as e:
            self.logger.error(f"[{rcs_task_no}] 任务异常: {e}")

    # ═══════════════════════════════════════════════════
    # 取消任务
    # ═══════════════════════════════════════════════════

    def cancel_task(self, body: dict) -> dict:
        task_code = body.get("taskCode", "")
        force = body.get("forceCancel", "1")

        # 通过 clientTaskCode 查找 RCS 任务
        rcs_task_no = self._find_by_client_code(task_code)
        if not rcs_task_no:
            return {"code": "-1", "msg": f"任务不存在: {task_code}"}

        task = self._tasks.get(rcs_task_no, {})
        agv_id = task.get("agvId", 0)

        # 取消定时器
        timer = self._timers.pop(rcs_task_no, None)
        if timer:
            timer.cancel()

        # 更新状态
        task["status"] = "CANCELLED"

        # 释放 AGV
        self.agv_mgr.set_idle(agv_id)

        self.logger.info(f"任务取消: {rcs_task_no}")
        return {"code": "0", "msg": "任务已取消"}

    # ═══════════════════════════════════════════════════
    # 继续任务
    # ═══════════════════════════════════════════════════

    def continue_task(self, body: dict) -> dict:
        task_code = body.get("taskCode", "")
        rcs_task_no = self._find_by_client_code(task_code)
        if not rcs_task_no:
            return {"code": "-1", "msg": f"任务不存在: {task_code}"}

        task = self._tasks.get(rcs_task_no, {})
        task["status"] = "RUNNING"

        # 重新启动 Worker
        timer = threading.Timer(1.0, self._task_worker, args=[rcs_task_no, task])
        self._timers[rcs_task_no] = timer
        timer.start()

        return {"code": "0", "msg": "任务已继续"}

    # ═══════════════════════════════════════════════════
    # 查询
    # ═══════════════════════════════════════════════════

    def query_task_status(self, task_codes: List[str]) -> dict:
        data = []
        for tc in task_codes:
            rcs_no = self._find_by_client_code(tc)
            if rcs_no and rcs_no in self._tasks:
                task = self._tasks[rcs_no]
                data.append({
                    "taskCode": task["clientTaskCode"],
                    "taskNo": task["rcsTaskNo"],
                    "taskStatus": task["status"],
                    "agvCode": task["agvCode"],
                })
        return {"code": "0", "msg": "ok", "data": data}

    def _find_by_client_code(self, client_code: str) -> str:
        """通过客户端任务码查找 RCS 任务号"""
        for rcs_no, task in self._tasks.items():
            if task.get("clientTaskCode") == client_code:
                return rcs_no
        return ""

    def summary(self) -> dict:
        with self._lock:
            running = sum(1 for t in self._tasks.values() if t["status"] == "RUNNING")
            completed = sum(1 for t in self._tasks.values() if t["status"] == "COMPLETED")
            return {
                "total": len(self._tasks),
                "running": running,
                "completed": completed,
                "tasks": {k: {
                    "rcsTaskNo": v["rcsTaskNo"],
                    "clientTaskCode": v["clientTaskCode"],
                    "status": v["status"],
                    "agvCode": v["agvCode"],
                    "srcBay": v["srcBay"],
                    "dstBay": v["dstBay"],
                } for k, v in list(self._tasks.items())[-10:]}
            }
```

### 3.5 `callback_sender.py` — 回调发送器

```python
#!/usr/bin/env python3
"""
回调发送器 — 向客户端推送 agvCallback / warnCallback
"""

import logging
import time
import requests
from typing import Dict, Any


class CallbackSender:
    """RCS → 客户端 HTTP 回调"""

    def __init__(self, base_url: str, logger: logging.Logger):
        self._base_url = base_url.rstrip('/')
        self._logger = logger

    # ── 任务状态回调 ──

    def send_agv_callback(self, taskCode: str, agvCode: str,
                          taskStatus: str, positionCode: str = ""):
        """
        发送 agvCallback
        taskStatus: 1=开始, 2=执行中, 9=已完成, 7=失败
        """
        url = f"{self._base_url}/agvCallback"
        payload = {
            "taskCode": taskCode,
            "agvCode": agvCode,
            "taskStatus": taskStatus,
            "positionCode": positionCode,
            "timestamp": time.strftime('%Y-%m-%d %H:%M:%S'),
        }
        self._post(url, payload, "agvCallback")

    # ── 告警回调 ──

    def send_warn_callback(self, warnCode: str, warnMsg: str,
                           agvCode: str = "", taskCode: str = "",
                           warnLevel: str = "1"):
        """发送 warnCallback"""
        url = f"{self._base_url}/warnCallback"
        payload = {
            "warnCode": warnCode,
            "warnMsg": warnMsg,
            "agvCode": agvCode,
            "taskCode": taskCode,
            "warnLevel": warnLevel,
            "timestamp": time.strftime('%Y-%m-%d %H:%M:%S'),
        }
        self._post(url, payload, "warnCallback")

    # ── 绑定通知回调 ──

    def send_bind_notify(self, stgBinCode: str, indBind: str, ctnrCode: str = ""):
        """发送 bindNotify"""
        url = f"{self._base_url}/bindNotify"
        payload = {
            "stgBinCode": stgBinCode,
            "indBind": indBind,
            "ctnrCode": ctnrCode,
            "timestamp": time.strftime('%Y-%m-%d %H:%M:%S'),
        }
        self._post(url, payload, "bindNotify")

    # ── 内部 ──

    def _post(self, url: str, payload: dict, label: str):
        try:
            resp = requests.post(url, json=payload, timeout=5)
            self._logger.info(
                f"[{label}] → {resp.status_code} "
                f"task={payload.get('taskCode','')[:20]} "
                f"status={payload.get('taskStatus','')}"
            )
        except requests.RequestException as e:
            self._logger.warning(f"[{label}] 回调发送失败 {url}: {e}")
```
