```python
    def get_exception_summary(self) -> dict:
        """获取异常日志汇总统计 (供 Web 仪表盘)"""
        try:
            with sqlite3.connect(self._db_path) as conn:
                conn.row_factory = sqlite3.Row

                # 各事件类型计数
                cursor = conn.execute(
                    "SELECT event_type, COUNT(*) as cnt FROM exception_log "
                    "GROUP BY event_type ORDER BY cnt DESC"
                )
                by_type = {row['event_type']: row['cnt'] for row in cursor.fetchall()}

                # 最近 24 小时异常数
                cursor = conn.execute(
                    "SELECT COUNT(*) as cnt FROM exception_log "
                    "WHERE logged_at > datetime('now', '-1 day')"
                )
                recent_24h = cursor.fetchone()['cnt']

                # 告警按等级分布
                cursor = conn.execute(
                    "SELECT alarm_level, COUNT(*) as cnt FROM alarm_logs "
                    "GROUP BY alarm_level"
                )
                by_alarm_level = {row['alarm_level']: row['cnt']
                                for row in cursor.fetchall()}

                return {
                    'total_exceptions': sum(by_type.values()),
                    'by_event_type': by_type,
                    'recent_24h': recent_24h,
                    'alarms': by_alarm_level,
                    'total_alarms': sum(by_alarm_level.values()),
                }
        except Exception:
            return {'error': '无法获取日志统计'}
```

## 2. P1-3：pickle 序列化上下文兼容性修复

### 2.1 问题分析

`task_instances.context_blob` 使用 `pickle.dumps(instance.context)`：

- Python 版本升级、类路径变化、context 字段结构变更都会导致反序列化失败；
- pickle 不是自描述格式，无法做字段级迁移；
- 恢复时反序列化失败会静默返回 `{}`，任务上下文（src/dst/robot/rcs_task_no）全部丢失。

### 2.2 实现目标

1. 新写入统一为 **JSON + schema_version 信封**，自描述、跨版本可读、可安全迁移；
2. 读取兼容旧 pickle 数据（先 JSON 后 pickle 兜底）；
3. 表结构增加 `context_schema` 列记录版本（旧库自动补列）；
4. 序列化/反序列化方法公开到引擎，供插件使用。

### 2.3 实施步骤与代码

#### 步骤 1：持久化层序列化改造

1a. 顶部引入 json 并新增信封常量与模块级工具函数：

```python
import sqlite3
import pickle
import json
import threading
import time
import logging
from typing import Dict, List, Any, Optional


CONTEXT_SCHEMA_VERSION = 1


def _serialize_context(context: dict) -> bytes:
    """上下文序列化：JSON + schema 版本信封（自描述，可迁移）"""
    try:
        payload = {
            'schema_version': CONTEXT_SCHEMA_VERSION,
            'data': context or {},
        }
        return json.dumps(payload, ensure_ascii=False, default=str).encode('utf-8')
    except Exception:
        # 兜底：不可 JSON 序列化的字段降级为 pickle（仅写入，不推荐长期使用）
        return pickle.dumps(context or {})


def _deserialize_context(blob) -> dict:
    """上下文反序列化：优先 JSON（新格式），兼容旧 pickle（历史数据）"""
    if blob is None:
        return {}
    if isinstance(blob, (bytes, bytearray)):
        raw = bytes(blob)
    else:
        raw = str(blob).encode('utf-8')

    # 1. JSON 信封（新格式）
    try:
        payload = json.loads(raw.decode('utf-8'))
        if isinstance(payload, dict) and payload.get('schema_version') == CONTEXT_SCHEMA_VERSION:
            data = payload.get('data')
            return data if isinstance(data, dict) else {}
    except Exception:
        pass

    # 2. pickle（旧格式兜底）
    try:
        data = pickle.loads(raw)
        return data if isinstance(data, dict) else {}
    except Exception:
        return {}
```

### 2.4 兼容与迁移说明

- 已入库的旧 pickle 数据在下次 `save()` 时会自动重写为 JSON 信封，无需手工迁移；
- 反序列化失败（JSON/pickle 均失败）返回 `{}` 并继续走恢复流程，任务会被标记人工介入，避免静默丢数据；
- 后续字段结构变更只需递增 `CONTEXT_SCHEMA_VERSION` 并编写版本迁移函数。

### 2.5 验收要点

- 新任务落库后 `context_blob` 为 JSON 信封（可用 `sqlite3` 直接查看）；
- 用旧 pickle 数据构造一条记录，重启后能正确恢复 src/dst/robot/rcs_task_no；
- 构造损坏 blob，恢复流程不崩溃并标记人工介入。

## 3. P1-6：SQLite 写放大修复

### 3.1 问题分析

当前每次状态转移的写路径：

```
WorkflowInstance._transition_to()
  ├─ ① persistence.log_state_change()   → 独立连接 + commit
  ├─ ② persistence.save()               → 独立连接 + commit（UPSERT）
  └─ ③ on_state_changed() → update_state() → 独立连接 + commit（重复更新）
```

一次状态转移产生 **3 次连接建立 + 3 次事务提交**；告警/异常日志又是独立连接。高并发（max_concurrent=10）下锁竞争明显。

### 3.2 实现目标

1. 状态变更日志 + 实例 UPSERT 合并为**单事务单连接**；
2. 去掉 `on_state_changed` 的重复写库（持久化职责收归 `save_with_log`）；
3. `update_state` 仅保留给恢复校准等"引擎外直写"场景。

### 3.3 实施步骤与代码

#### 步骤 1：持久化层新增单事务写入

```python
                      trigger_event: str = '') -> bool:
```

## 4. P1-2：多缓存状态一致性修复

### 4.1 问题分析

任务级状态由多个插件分散写入：

| 写操作 | 当前位置 |
|---|---|
| AGV 忙碌/空闲 | `task_queue.try_dequeue`（set_busy）、`workflow_engine_plugin._action_release_resources`（set_idle）、`status_poller`（轮询覆盖） |
| 起点仓位 in_task / bind_status | `task_queue`、`trigger_engine.scan/release`、`workflow_engine_plugin`、`callback_handler`（bindNotify） |
| 终点仓位 is_empty | `status_poller`（polling）、`callback_handler`（rcs_callback）、`workflow_engine_plugin`（task_complete） |
| 资源锁 | `trigger_engine`、`task_queue`、`workflow_engine_plugin` |

风险：轮询旧数据覆盖新状态、释放路径不一致（release 时机/字段不统一）、无版本可观测性。

### 4.2 实现目标

1. 新增 `StateCoordinator`：`dispatch_task / release_task / get_snapshot` 三个原子化入口，内部串行化"锁 + 缓存"级联写；
2. 缓存条目增加 `revision`（单调递增）与 `source`（写入来源），提升可观测性并支持后续"旧写拒绝"；
3. 物理状态（轮询）与业务状态（任务）写入入口分离：轮询只更新物理字段；
4. 工作流释放、触发器释放统一走 coordinator（幂等）。

#### 步骤 2：新增 StateCoordinator

新增文件：`app_mgr_object/components/task_cache/state_coordinator.py`

```python
#!/usr/bin/env python3
"""
状态协调器 — 统一资源状态联动更新入口（P1-2）

背景：RobotCache / BayCache / DestBayCache / ResourceLocker / TaskQueue
的状态此前由多个插件分散写入，缺少统一入口与版本校验，存在"旧轮询覆盖
新状态"的竞态风险。本协调器把"任务生命周期"级联写操作收敛为：
  - dispatch_task()  下发时的资源占用（锁 + 起点仓位在任务 + AGV 忙碌）
  - release_task()   终态时的资源释放（锁 + 仓位释放 + AGV 空闲 + 终点置货）
所有任务级写操作在协调器内部加锁串行化。
"""

import threading
import logging


class StateCoordinator:
    """资源状态协调器（组装各缓存，提供原子化联动操作）"""

    def __init__(self, robot_cache, bay_cache, dest_cache, locker,
                 logger: logging.Logger = None):
        self._robot_cache = robot_cache
        self._bay_cache = bay_cache
        self._dest_cache = dest_cache
        self._locker = locker
        self._logger = logger or logging.getLogger(__name__)
        self._lock = threading.RLock()

    # ── 下发：资源占用 ──
    def dispatch_task(self, robot_id: int, src_bay: str, dst_bay: str,
                      task_id: str = '') -> bool:
        """统一执行：锁定资源 + 标记起点仓位在任务中 + 标记 AGV 忙碌。

        Returns:
            True  = 资源占用成功（调用方可以继续下发）
            False = 资源冲突（不应继续下发）
        """
        with self._lock:
            if not self._locker.lock(robot_id, src_bay, dst_bay):
                self._log_debug(
                    f"dispatch 资源冲突: robot={robot_id} src={src_bay} dst={dst_bay}"
                )
                return False
            if src_bay:
                self._bay_cache.mark_in_task(src_bay, True, source='dispatch')
            if robot_id:
                self._robot_cache.set_busy(robot_id, task_id or src_bay,
                                           source='dispatch')
            self._log_info(f"dispatch: robot={robot_id} {src_bay}→{dst_bay} task={task_id}")
            return True

    # ── 终态：资源释放（幂等） ──
    def release_task(self, task_context: dict) -> None:
        """统一执行：释放资源锁 + 起点仓位出任务 + AGV 空闲 + 终点仓位置货"""
        with self._lock:
            robot_id = task_context.get('robot_id')
            src_bay = task_context.get('src_bay')
            dst_bay = task_context.get('dst_bay')

            self._locker.release(
                robot_id=robot_id,
                src_bay=src_bay,
                dst_bay=dst_bay,
            )
            if src_bay:
                self._bay_cache.mark_in_task(src_bay, False, source='release')
            if robot_id:
                self._robot_cache.set_idle(robot_id, source='release')
            if dst_bay:
                self._dest_cache.update(dst_bay, is_empty=False,
                                        source='task_complete')
            self._log_info(f"release: robot={robot_id} src={src_bay} dst={dst_bay}")

    def get_snapshot(self) -> dict:
        """汇总状态快照（供 Web/运维）"""
        with self._lock:
            return {
                'locked': self._locker.get_locked_count(),
                'robots': self._robot_cache.get_all_robot_status(),
                'src_bays': self._bay_cache.get_all_bay_status(),
                'dst_bays': self._dest_cache.get_all_bay_status(),
            }

    def _log_info(self, msg: str):
        if self._logger:
            self._logger.info(f"[StateCoordinator] {msg}")

    def _log_debug(self, msg: str):
        if self._logger:
            self._logger.debug(f"[StateCoordinator] {msg}")
```

```python
    def try_dequeue(self):
        """尝试出队一个任务 (需 AGV 空闲 + 资源未锁定)"""
        with self._lock:
            self._age_out_locked()          # P1-5：先老化
            if not self._queue:

                    ok = self._state_coordinator.dispatch_task(
                        robot_id, candidate.src_bay, candidate.dst_bay,
                        task_id='',

                        self._robot_cache.set_busy(robot_id, candidate.src_bay)

                            f"(剩余队列={len(self._queue)})"
                        )
                    return (candidate, robot_id)

            self._log_debug("队列任务与当前锁定资源冲突, 等待释放")
```

```python
                # 兼容降级：原分散释放逻辑（smart_trigger 未激活时兜底）
                src_bay = context.get('src_bay')
                dst_bay = context.get('dst_bay')
                robot_id = context.get('robot_id')

                status_poller = getattr(self.node, 'status_poller', None)
                if status_poller and robot_id:
                    status_poller.robot_cache.set_idle(robot_id)
                if src_bay:
                    bay_fusion = getattr(self.node, 'bay_status_fusion', None)
                    if bay_fusion and bay_fusion.cache:
                        bay_fusion.cache.set_unbound(src_bay)
                        bay_fusion.cache.mark_in_task(src_bay, False)
                if dst_bay and status_poller and status_poller.dest_cache:
                    smart_trigger = getattr(self.node, 'smart_trigger', None)
                    if smart_trigger and getattr(smart_trigger, '_trigger_engine', None):
                        smart_trigger._trigger_engine.release_dst_bay(dst_bay)
                    status_poller.dest_cache.update(
                        dst_bay, is_empty=False, source='task_complete'
                    )

            self._publish_event('workflow.resources_released', {
                'task_id': context.get('task_id'),
                'src_bay': context.get('src_bay'),
                'robot_id': context.get('robot_id'),
            })
            return 'done'
        except Exception as e:
            self.logger.error(f"释放资源异常: {e}")
            return 'done'
```

> 注意：`coordinator` 在 smart_trigger 激活后才挂载；工作流 Action 在运行期执行，此时 coordinator 已就绪。保留降级分支防止启动顺序变化。

## 5. P1-5：队列无老化机制修复

### 5.1 问题分析

`TaskQueue` 中候选任务一旦因 AGV 全忙或资源冲突无法出队，会**永久滞留**，且每次扫描被排除（`exclude_src/dst`）不再重复入队，最终形成僵尸候选；`max_queue_size` 配置在 smart_trigger 中读取但从未真正生效。

### 5.2 实现目标

1. 候选任务记录入队时间戳（`enqueue_time`）；
2. TTL 老化：超时未出队自动移除（下次扫描若条件仍满足会自动重新生成）；
3. 队列容量上限：满时拒绝新候选并计数；
4. 统计项：`aged_out_count / skip_full_count / oldest_wait`。

### 5.3 实施步骤与代码

2b. 新增老化方法（内部方法由持有锁的调用方使用，避免重复加锁死锁）：

```python
    def age_out(self) -> int:
        """移除超时未出队的候选任务（公开入口，独立加锁）"""
        with self._lock:
            return self._age_out_locked()

    def _age_out_locked(self) -> int:
        if not self._queue:
            return 0
        now = time.time()
        keep = []
        removed = 0
        for c in self._queue:
            if c.enqueue_time and (now - c.enqueue_time) > self._aging_ttl:
                removed += 1
                self._queued_src.discard(c.src_bay)
                self._queued_dst.discard(c.dst_bay)
            else:
                keep.append(c)
        if removed:
            self._queue = keep
            self._aged_out_count += removed
            if self._logger:
                self._logger.info(
                    f"任务老化移除 {removed} 个 (TTL={self._aging_ttl}s)"
                )
        return removed
```

2c. `get_stats` 增加老化统计：

```python
    def get_stats(self) -> dict:
        with self._lock:
            oldest_wait = 0.0
            now = time.time()
            for c in self._queue:
                if c.enqueue_time:
                    oldest_wait = max(oldest_wait, now - c.enqueue_time)
            return {
                'queue_size': len(self._queue),
                'enqueue_count': self._enqueue_count,
                'dequeue_count': self._dequeue_count,
                'skip_duplicate_count': self._skip_duplicate_count,
                'aged_out_count': self._aged_out_count,
                'skip_full_count': self._skip_full_count,
                'oldest_wait': oldest_wait,
                'queued_src': list(self._queued_src),
                'queued_dst': list(self._queued_dst),
            }
```
