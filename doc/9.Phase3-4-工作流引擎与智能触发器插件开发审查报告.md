### 2.5 persistence.py 审查及缺陷修复

#### 缺陷 1: `datetime_now()` 在 `save()` INSERT 中未被正确引用

```python
# 当前代码（有缺陷）：
def save(self, instance):
    conn.execute("""
        INSERT OR REPLACE INTO task_instances
        (task_id, workflow_name, src_bay, dst_bay, robot_id,
         cargo_type, rcs_task_no, status, current_state, context_blob)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
    """, (
        ...
        self._status_from_state(instance.current_state),
        ...,
        context_blob
    ))
    # 问题：created_at 不在 SQL INSERT 中，但 OR REPLACE 会替换整行
    # 导致 created_at 被重置为 DEFAULT 值，丢失原始创建时间
```

**修复**: 使用 `INSERT ... ON CONFLICT ... DO UPDATE` 或先查询再 INSERT。

#### 缺陷 2: `update_state` 中引用了未导入的 `datetime_now`

```python
def update_state(self, task_id, state):
    conn.execute("""
        UPDATE task_instances
        SET current_state = ?, status = ?, completed_at = ?
        WHERE task_id = ?
    """, (
        state,
        status,
        datetime_now() if state in ('completed', 'failed') else None,  # ← datetime_now 未导入!
        task_id
    ))
```

**修复**: `datetime_now()` 定义在模块作用域（line ~180），引用没问题——但如果在 `update_state` 所在作用域没有该名称，会报错。确认模块底部有 `def datetime_now()`，实际上可解决。但为了防止导入问题，建议改为类方法。

#### 修复后代码（persistence.py 修改部分）

```python
# ═════════════════════════════════════════════════════════════
# persistence.py 修复版 — 仅列出修改的方法
# ═════════════════════════════════════════════════════════════

def save(self, instance) -> bool:
    """保存工作流实例（使用 INSERT OR REPLACE 但保留 created_at）"""
    with self._lock:
        try:
            with sqlite3.connect(self._db_path) as conn:
                context_blob = pickle.dumps(instance.context)
                now = _now_str()
                conn.execute("""
                    INSERT INTO task_instances
                    (task_id, workflow_name, src_bay, dst_bay, robot_id,
                     cargo_type, rcs_task_no, status, current_state, context_blob, created_at)
                    VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
                    ON CONFLICT(task_id) DO UPDATE SET
                        workflow_name = excluded.workflow_name,
                        src_bay       = excluded.src_bay,
                        dst_bay       = excluded.dst_bay,
                        robot_id      = excluded.robot_id,
                        cargo_type    = excluded.cargo_type,
                        rcs_task_no   = excluded.rcs_task_no,
                        status        = excluded.status,
                        current_state = excluded.current_state,
                        context_blob  = excluded.context_blob,
                        completed_at  = ?
                """, (
                    instance.id,
                    instance.definition.name,
                    instance.context.get('src_bay', ''),
                    instance.context.get('dst_bay', ''),
                    instance.context.get('robot_id', 0),
                    instance.context.get('cargo_type', 0),
                    instance.context.get('rcs_task_no', ''),
                    _status_from_state(instance.current_state),
                    instance.current_state,
                    context_blob,
                    now,         # created_at (首次插入时使用)
                    now if instance.definition.is_final_state(instance.current_state)
                    else None,   # completed_at (仅在终态时写入)
                ))
                conn.commit()
            return True
        except Exception as e:
            self._log_error(f"保存实例失败 {instance.id}: {e}")
            return False


def update_state(self, task_id: str, state: str):
    """更新实例当前状态"""
    with self._lock:
        try:
            with sqlite3.connect(self._db_path) as conn:
                conn.execute("""
                    UPDATE task_instances
                    SET current_state = ?, status = ?, completed_at = ?
                    WHERE task_id = ?
                """, (
                    state,
                    _status_from_state(state),
                    _now_str() if state in ('completed', 'failed') else None,
                    task_id
                ))
                conn.commit()
        except Exception as e:
            self._log_error(f"更新状态失败 {task_id}: {e}")


# ── 模块级工具函数（放在文件末尾） ──

def _now_str() -> str:
    """获取当前时间字符串"""
    return time.strftime('%Y-%m-%d %H:%M:%S', time.localtime())


def _status_from_state(state: str) -> str:
    """从工作流状态映射到任务状态"""
    mapping = {
        'calling_rcs':         'DISPATCHING',
        'monitoring':          'RUNNING',
        'compensating_query':  'RUNNING',
        'releasing':           'RUNNING',
        'completed':           'SUCCESS',
        'failed':              'FAILED',
        'manual_intervention': 'MANUAL',
    }
    return mapping.get(state, 'RUNNING')
```

> **注意**: 修复后应将 `_now_str` 和 `_status_from_state` 转为模块级函数（非类方法），确保 `update_state` 和 `save` 中均可访问。原有的 `datetime_now()` 和 `_status_from_state` 改为新名称。

### 2.6 event_bridge.py 审查及缺陷修复

#### 缺陷: `_on_rcs_agv_callback` 中字段映射错误

```python
# 当前代码（有缺陷）：
def _on_rcs_agv_callback(self, data):
    task_code = data.get('task_code', '')
    method = data.get('method', '')       # ← callback_handler 发布的事件中没有 'method' 字段!
    if task_code and self._engine:
        event = 'callback_completed' if method == 'end' else 'callback_exception'
        ...
```

**问题**: `callback_handler_plugin._on_agv_callback()` 发布的 `rcs.agv_callback` 事件 data 结构为：
```python
{'task_code': '...', 'agv_code': '...', 'task_status': '1', 'position_code': '...', 'timestamp': '...'}
```

其中**没有 `method` 字段**。正确的任务状态字段是 `task_status`。

**RCS 任务状态码映射**（来自 `RCSTaskStatus` 常量）:
- `"2"` (RUNNING) → 执行中 → `callback_running`
- `"9"` (COMPLETED) → 已完成 → `callback_completed`
- `"7"` (FAILED) → 失败 → `callback_exception`

#### 修复后代码（event_bridge.py 修改部分）

```python
def _on_rcs_agv_callback(self, data: dict):
    """RCS agvCallback → 注入工作流实例"""
    task_code = data.get('task_code', '')
    task_status = data.get('task_status', '')

    if not task_code or not self._engine:
        return

    # 映射 task_status → 工作流事件名
    status_event_map = {
        '9': 'callback_completed',   # COMPLETED
        '7': 'callback_exception',   # FAILED
        '8': 'callback_exception',   # CANCELLED
    }
    event = status_event_map.get(task_status, 'callback_running')

    self._log_debug(f"RCScallback → workflow: task={task_code} "
                    f"status={task_status} → event={event}")
    asyncio.ensure_future(
        self._engine.inject_event(task_code, event, data)
    )
```

### 5.2 完整代码

**路径**: `app_mgr_object/plugins/workflow_engine_plugin.py`

```python
#!/usr/bin/env python3
"""
workflow_engine_plugin — 工作流执行引擎插件

职责：
  - 管理 WorkflowEngine 生命周期
  - 注册 8 个 Action 函数（invoke_gen_task / wait_for_completion ...）
  - 连接 EventBus ↔ WorkflowEngine 事件桥
  - 订阅 trigger.task_created 事件，启动工作流实例
  - 启动时恢复未完成实例（断电恢复）

依赖：
  - rcs_adapter (Phase 1)
  - EventBus
  - WorkflowEngine / WorkflowEventBridge / 全部 workflow 组件
"""

import asyncio
import os
from typing import Dict, Any

from .base_plugin import BasePlugin
from app_mgr_object.components.workflow import (
    WorkflowEngine,
    WorkflowEventBridge,
)


class WorkflowEnginePlugin(BasePlugin):
    """工作流执行引擎插件"""

    PLUGIN_NAME = "workflow_engine"
    PLUGIN_VERSION = "1.0.0"

    def __init__(self, node, config: Dict[str, Any] = None):
        super().__init__(node, config)

        # 配置
        pkg_share = self.config.get('pkg_share_dir', '')
        self._definitions_dir = self.config.get(
            'definitions_dir',
            os.path.join(pkg_share, 'config', 'workflows')
        )
        self._db_path = self.config.get(
            'db_path',
            os.path.join(pkg_share, 'config', 'task_instances.db')
        )
        self._max_concurrent = self.config.get('max_concurrent', 10)
        self._scan_interval = self.config.get('scan_interval', 2.0)

        # 引擎组件
        self._engine: WorkflowEngine = None
        self._event_bridge: WorkflowEventBridge = None

        # 状态
        self._running = False
        self._task: asyncio.Task = None

    # ═════════════════════════════════════════════════════════
    #  生命周期
    # ═════════════════════════════════════════════════════════

    def _configure_impl(self) -> bool:
        """初始化 WorkflowEngine，加载 YAML 定义，注册 Action"""
        try:
            # 创建引擎
            self._engine = WorkflowEngine(
                definitions_dir=self._definitions_dir,
                db_path=self._db_path,
                max_concurrent=self._max_concurrent,
                logger=self.logger,
            )

            # 加载工作流定义
            count = self._engine.load_definitions()
            self.logger.info(f"加载 {count} 个工作流定义")

            # 注册 8 个 Action 函数
            self._register_actions()

            # 创建事件桥
            self._event_bridge = WorkflowEventBridge(logger=self.logger)

            # 挂载到节点
            self.node.workflow_engine = self

            self.logger.info(
                f"WorkflowEngine 配置完成: max_concurrent={self._max_concurrent}, "
                f"defs={count}"
            )
            return True

        except Exception as e:
            self.logger.error(f"WorkflowEngine 配置失败: {e}")
            return False

    def _activate_impl(self) -> bool:
        """激活：恢复实例 + 连接 EventBus + 订阅事件"""
        loop = asyncio.get_event_loop()

        # 1. 恢复未完成实例（断电恢复）
        loop.create_task(self._engine.recover_all())

        # 2. 连接 EventBus → Engine 事件桥
        self._event_bridge.connect(self._engine, self.event_bus)
        self._event_bridge.subscribe_rcs_events()

        # 3. 订阅 trigger.task_created 事件
        self.event_bus.subscribe('trigger.task_created', self._on_trigger_task_created)

        self._running = True
        self.logger.info("WorkflowEngine 激活完成")
        return True

    def _deactivate_impl(self) -> bool:
        self._running = False
        return True

    def _cleanup_impl(self) -> bool:
        self._engine = None
        self._event_bridge = None
        return True

    # ═════════════════════════════════════════════════════════
    #  Action 注册
    # ═════════════════════════════════════════════════════════

    def _register_actions(self):
        """注册全部 Action 函数到引擎"""
        self._engine.register_actions({
            'invoke_gen_task':          self._action_invoke_gen_task,
            'wait_for_completion':      self._action_wait_for_completion,
            'compensate_query':         self._action_compensate_query,
            'release_resources':        self._action_release_resources,
            'publish_completed_event':  self._action_publish_completed,
            'publish_failed_event':     self._action_publish_failed,
            'publish_manual_event':     self._action_publish_manual,
            'log_error':                self._action_log_error,
        })
        self.logger.info("已注册 8 个 Action 函数")

    # ═════════════════════════════════════════════════════════
    #  Action 实现
    # ═════════════════════════════════════════════════════════

    async def _action_invoke_gen_task(self, context: dict, instance) -> str:
        """调用 RCS genAgvSchedulingTask 下发任务

        Returns:
            "rcs_ack_ok"   — RCS 返回成功
            "rcs_ack_fail" — RCS 返回失败
        """
        self.logger.info(f"下发任务: {context.get('task_id')} "
                         f"src={context.get('src_bay')} dst={context.get('dst_bay')}")

        try:
            rcs = self._get_rcs_adapter()
            resp = rcs.gen_task(context)

            if resp.get('code') == '0':
                # 记录 RCS 任务流水号
                rcs_task_no = resp.get('data', {}).get('taskNo', '')
                if rcs_task_no:
                    context['rcs_task_no'] = rcs_task_no
                self.logger.info(f"任务下发成功: {context['task_id']} "
                                 f"→ RCS taskNo={rcs_task_no}")
                return 'rcs_ack_ok'
            else:
                self.logger.error(f"任务下发失败: {resp.get('msg', '未知错误')}")
                return 'rcs_ack_fail'

        except Exception as e:
            self.logger.error(f"任务下发异常: {e}")
            return 'rcs_ack_fail'

    async def _action_wait_for_completion(self, context: dict, instance) -> str:
        """挂起等待 agvCallback 回调完成

        此 Action 不会主动返回 —— instance 在 monitoring 状态进入时
        通过 `wait_for_completion` action 调用此函数。
        实际的事件通过 `inject_event` 从 event_bridge 注入 instance 的 event_queue。
        这里的实现只需要返回一个标记，让 instance 知道进入等待模式。

        Returns:
            "waiting" — 通知 instance 进入挂起等待模式
        """
        return 'waiting'

    async def _action_compensate_query(self, context: dict, instance) -> str:
        """主动查询 RCS 任务状态（超时补偿）

        Returns:
            "rcs_completed" — RCS 任务已结束
            "rcs_running"   — RCS 任务仍在执行
            "rcs_failed"    — RCS 任务失败
        """
        rcs_task_no = context.get('rcs_task_no', '')
        if not rcs_task_no:
            self.logger.warning(f"无 RCS 任务号，无法补偿查询: {context['task_id']}")
            return 'rcs_running'

        self.logger.info(f"补偿查询: {context['task_id']} → RCS {rcs_task_no}")

        try:
            rcs = self._get_rcs_adapter()
            resp = rcs.query_task_status([rcs_task_no])

            if resp.get('code') != '0':
                return 'rcs_running'

            data_list = resp.get('data', [])
            if not data_list:
                return 'rcs_running'

            task_status = str(data_list[0].get('taskStatus', ''))

            if task_status == '9':
                self.logger.info(f"补偿查询确认完成: {context['task_id']}")
                return 'rcs_completed'
            elif task_status in ('7', '8'):
                self.logger.warning(f"补偿查询确认失败: {context['task_id']}")
                return 'rcs_failed'
            else:
                self.logger.info(f"补偿查询: 任务仍在执行: {context['task_id']}")
                return 'rcs_running'

        except Exception as e:
            self.logger.error(f"补偿查询异常: {e}")
            return 'rcs_running'

    async def _action_release_resources(self, context: dict, instance) -> str:
        """释放资源（叉车、仓位标记）"""
        self.logger.info(f"释放资源: {context.get('task_id')}")

        try:
            # 通知 smart_trigger 释放资源
            self._publish_event('workflow.resources_released', {
                'task_id': context.get('task_id'),
                'src_bay': context.get('src_bay'),
                'robot_id': context.get('robot_id'),
            })

            # 更新 RobotCache（标记叉车为空闲）
            status_poller = getattr(self.node, 'status_poller', None)
            robot_id = context.get('robot_id')
            if status_poller and robot_id:
                status_poller.robot_cache.set_idle(robot_id)

            return 'done'

        except Exception as e:
            self.logger.error(f"释放资源异常: {e}")
            return 'done'

    async def _action_publish_completed(self, context: dict, instance) -> str:
        """发布任务完成事件"""
        self._publish_event('workflow.task_completed', {
            'task_id': context.get('task_id'),
            'src_bay': context.get('src_bay'),
            'dst_bay': context.get('dst_bay'),
            'robot_id': context.get('robot_id'),
            'duration': instance.duration,
        })
        return 'done'

    async def _action_publish_failed(self, context: dict, instance) -> str:
        """发布任务失败事件"""
        self._publish_event('workflow.task_failed', {
            'task_id': context.get('task_id'),
            'state': instance.current_state,
            'src_bay': context.get('src_bay'),
            'dst_bay': context.get('dst_bay'),
            'robot_id': context.get('robot_id'),
        })
        return 'done'

    async def _action_publish_manual(self, context: dict, instance) -> str:
        """发布人工介入事件"""
        self._publish_event('workflow.task_needs_manual', {
            'task_id': context.get('task_id'),
            'state': instance.current_state,
            'reason': '需要人工介入处理',
        })
        return 'done'

    async def _action_log_error(self, context: dict, instance) -> str:
        """记录错误日志"""
        self.logger.error(
            f"工作流错误: {context.get('task_id')} "
            f"state={instance.current_state}"
        )
        return 'done'

    # ═════════════════════════════════════════════════════════
    #  事件订阅
    # ═════════════════════════════════════════════════════════

    def _on_trigger_task_created(self, data: dict):
        """接收 SmartTrigger 生成的任务 → 创建工作流实例"""
        asyncio.ensure_future(
            self._engine.create_instance(
                workflow_name='agv_transport_atomic',
                context=data,
            )
        )
        self.logger.info(f"收到触发器任务: {data.get('task_id')}")

    # ═════════════════════════════════════════════════════════
    #  公共查询接口（Web 运维用）
    # ═════════════════════════════════════════════════════════

    def get_active_task_count(self) -> int:
        return self._engine.get_active_count() if self._engine else 0

    def get_all_tasks(self, limit: int = 100) -> list:
        return self._engine.get_all_instances_from_db(limit) if self._engine else []

    def cancel_task(self, task_id: str):
        """Web 手动取消任务"""
        if self._engine:
            asyncio.ensure_future(self._engine.cancel_instance(task_id))

    # ═════════════════════════════════════════════════════════
    #  辅助
    # ═════════════════════════════════════════════════════════

    def _get_rcs_adapter(self):
        adapter = getattr(self.node, 'rcs_adapter', None)
        if adapter is None:
            adapter = self.node.plugin_manager.get_plugin('rcs_adapter')
        if adapter is None:
            raise RuntimeError("RCS 适配器未加载")
        return adapter

    def get_status(self) -> Dict[str, Any]:
        base = super().get_status()
        base.update({
            'active_tasks': self.get_active_task_count(),
            'max_concurrent': self._max_concurrent,
            'definitions_dir': self._definitions_dir,
            'db_path': self._db_path,
            'running': self._running,
        })
        return base
```

### 6.2 完整代码

**路径**: `app_mgr_object/plugins/smart_trigger_plugin.py`

```python
#!/usr/bin/env python3
"""
smart_trigger_plugin — 智能触发器插件

职责：
  - 定时扫描多源缓存（起始仓位/终点仓位/叉车状态）
  - 匹配条件：绑定仓位 + 空终点仓位 + 空闲叉车 + 工作时间窗口
  - 生成 TaskContext 并通过 EventBus 发布 trigger.task_created
  - 订阅 bay.bound / workflow.task_completed 事件加速触发

依赖：
  - bay_status_fusion (Phase 2)
  - status_poller (Phase 2)
  - workflow_engine (Phase 3)
  - TriggerEngine / ScheduleChecker / ResourceLocker (组件层)
"""

import asyncio
from typing import Dict, Any, Optional

from .base_plugin import BasePlugin
from app_mgr_object.components.trigger import (
    TriggerEngine,
    ScheduleChecker,
)
from app_mgr_object.components.task_cache.config_cache import ConfigCache


class SmartTriggerPlugin(BasePlugin):
    """智能触发器插件 — 条件扫描 + 任务生成"""

    PLUGIN_NAME = "smart_trigger"
    PLUGIN_VERSION = "1.0.0"

    def __init__(self, node, config: Dict[str, Any] = None):
        super().__init__(node, config)

        # 扫描间隔
        self._scan_interval = self.config.get('scan_interval', 2.0)
        self._enabled = self.config.get('enabled', True)

        # 引擎组件
        self._trigger_engine: Optional[TriggerEngine] = None
        self._config_cache: Optional[ConfigCache] = None

        # 状态
        self._running = False
        self._scan_task: Optional[asyncio.Task] = None
        self._scan_count: int = 0
        self._trigger_count: int = 0

    # ═════════════════════════════════════════════════════════
    #  生命周期
    # ═════════════════════════════════════════════════════════

    def _configure_impl(self) -> bool:
        """初始化 TriggerEngine（依赖 Phase 2 插件已加载）"""
        try:
            # 获取 Phase 2 插件的缓存实例
            bay_fusion = self._get_plugin('bay_status_fusion')
            status_poller = self._get_plugin('status_poller')

            if not bay_fusion or not bay_fusion.cache:
                raise RuntimeError("bay_status_fusion 插件或其 cache 未就绪")
            if not status_poller or not status_poller.dest_cache:
                raise RuntimeError("status_poller 插件或其 cache 未就绪")

            # 初始化配置缓存
            self._config_cache = ConfigCache(logger=self.logger)

            # 创建触发器引擎
            self._trigger_engine = TriggerEngine(
                bay_cache=bay_fusion.cache,
                dest_cache=status_poller.dest_cache,
                robot_cache=status_poller.robot_cache,
                config_cache=self._config_cache,
                logger=self.logger,
            )

            self.logger.info(
                f"SmartTrigger 配置完成: scan_interval={self._scan_interval}s"
            )
            return True

        except Exception as e:
            self.logger.error(f"SmartTrigger 配置失败: {e}")
            return False

    def _activate_impl(self) -> bool:
        """激活：启动 asyncio 扫描循环 + 订阅事件"""
        if not self._enabled:
            self.logger.info("SmartTrigger 已禁用")
            return True

        # 启动定时扫描
        loop = asyncio.get_event_loop()
        self._scan_task = loop.create_task(self._scan_loop())

        # 订阅仓位绑定事件（加速触发）
        self.event_bus.subscribe('bay.bound', self._on_bay_bound)

        # 订阅任务完成/失败事件（释放资源 + 加速下一轮扫描）
        self.event_bus.subscribe('workflow.task_completed',
                                 self._on_task_completed)
        self.event_bus.subscribe('workflow.task_failed',
                                 self._on_task_failed)
        self.event_bus.subscribe('workflow.resources_released',
                                 self._on_resources_released)

        self._running = True
        self.logger.info(f"SmartTrigger 激活完成, 扫描间隔={self._scan_interval}s")
        return True

    def _deactivate_impl(self) -> bool:
        self._running = False
        if self._scan_task:
            self._scan_task.cancel()
        return True

    def _cleanup_impl(self) -> bool:
        self._trigger_engine = None
        self._config_cache = None
        return True

    # ═════════════════════════════════════════════════════════
    #  扫描循环
    # ═════════════════════════════════════════════════════════

    async def _scan_loop(self):
        """主扫描循环：定时遍历 6 种货物类型"""
        self.logger.info("扫描循环启动")

        while self._running:
            try:
                if self._trigger_engine._schedule_checker.is_in_working_window():
                    # 扫描所有货物类型
                    for cargo_type in range(1, 7):
                        ctx = self._trigger_engine.scan(cargo_type)
                        if ctx:
                            self._scan_count += 1
                            self._trigger_count += 1
                            # 发布任务创建事件 → workflow_engine 订阅并创建工作流
                            self._publish_event('trigger.task_created', ctx.to_dict())
                            self.logger.info(
                                f"触发任务 #{self._trigger_count}: {ctx.task_id} "
                                f"src={ctx.src_bay} dst={ctx.dst_bay} "
                                f"robot={ctx.robot_id} type={cargo_type}"
                            )
                else:
                    self.logger.debug("不在工作时间窗口，跳过扫描")

            except Exception as e:
                self.logger.error(f"扫描循环异常: {e}")

            await asyncio.sleep(self._scan_interval)

    async def _scan_once(self):
        """单次扫描（由事件触发时调用）"""
        self.logger.debug("事件触发单次扫描")
        try:
            for cargo_type in range(1, 7):
                ctx = self._trigger_engine.scan(cargo_type)
                if ctx:
                    self._scan_count += 1
                    self._trigger_count += 1
                    self._publish_event('trigger.task_created', ctx.to_dict())
        except Exception as e:
            self.logger.error(f"单次扫描异常: {e}")

    # ═════════════════════════════════════════════════════════
    #  事件回调
    # ═════════════════════════════════════════════════════════

    def _on_bay_bound(self, data: dict):
        """仓位绑定完成 → 立即触发一次扫描"""
        bay_id = data.get('bay_id', '')
        self.logger.info(f"收到绑定事件: {bay_id}，加速扫描")
        asyncio.ensure_future(self._scan_once())

    def _on_task_completed(self, data: dict):
        """任务完成 → 释放资源 + 立即扫描"""
        task_id = data.get('task_id', '')
        self.logger.info(f"收到任务完成事件: {task_id}，释放资源并扫描")
        self._trigger_engine.release_task_resources(data)
        asyncio.ensure_future(self._scan_once())

    def _on_task_failed(self, data: dict):
        """任务失败 → 释放资源 + 立即扫描"""
        task_id = data.get('task_id', '')
        self.logger.warning(f"收到任务失败事件: {task_id}，释放资源并扫描")
        self._trigger_engine.release_task_resources(data)
        asyncio.ensure_future(self._scan_once())

    def _on_resources_released(self, data: dict):
        """资源释放通知 → 更新缓存"""
        self._trigger_engine.release_task_resources(data)

    # ═════════════════════════════════════════════════════════
    #  辅助
    # ═════════════════════════════════════════════════════════

    def _get_plugin(self, name: str):
        """获取插件实例"""
        plugin = getattr(self.node, name, None)
        if plugin is None:
            plugin = self.node.plugin_manager.get_plugin(name)
        return plugin

    def get_status(self) -> Dict[str, Any]:
        base = super().get_status()
        base.update({
            'enabled': self._enabled,
            'scan_interval': self._scan_interval,
            'running': self._running,
            'scan_count': self._scan_count,
            'trigger_count': self._trigger_count,
            'locked_resources': self._trigger_engine.get_locked_count()
                if self._trigger_engine else {},
        })
        return base
```
