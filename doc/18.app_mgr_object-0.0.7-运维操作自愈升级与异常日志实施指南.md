## 2. 需求 3: ROS2 Service /task_control — 运维手动操作接口

### 2.1 设计

**服务名**: `/task_control`  
**服务定义**: `app_mgr_object/srv/TaskControl.srv`

支持 4 种操作:
- `force_trigger` — 强制下发搬运任务 (指定 src/dst/robot)
- `cancel` — 取消指定任务
- `retry` — 重试指定任务 (从 manual_intervention 退回 calling_rcs)
- `query` — 查询指定任务状态

### 2.2 Step 9-1: 新建服务定义文件

**文件路径**: `app_mgr_object/srv/TaskControl.srv` **(新建)**

```
# TaskControl.srv — 任务控制运维操作接口
# 供 Web 面板 / 命令行工具调用

string operation     # force_trigger | cancel | retry | query
string task_id       # 目标任务 ID (query 时可选, 其他操作必填)
# ---------- force_trigger 专用字段 ----------
string src_bay       # 起始仓位
string dst_bay       # 终点仓位
int32  robot_id      # AGV 编号
int32  cargo_type    # 货物类型 1~6
---
bool   success       # 操作是否成功
string message       # 返回消息 (错误原因或状态描述)
string task_status   # 任务当前状态 (query 时返回)
string current_state # 工作流当前状态 (query 时返回)
string task_id_out   # 返回的任务 ID (force_trigger 时为新生成的 task_id)
```

```python
    """激活：恢复实例 + 连接 EventBus + 订阅事件"""
    self._loop = self.node.asyncio_loop

    # 1. 恢复前 RCS 状态校准

        self._validate_and_recover(), self._loop

    # 2. 连接 EventBus → Engine 事件桥
```

**Step C — 新增 `_on_task_control` 回调方法**（放在 `cancel_task` 方法之后）:

```python
# ====== 新增: ROS2 Service /task_control 回调 ======

def _on_task_control(self, request, response):
    """处理运维操作请求

    支持操作:
      - force_trigger: 强制下发任务
      - cancel:        取消任务
      - retry:         重试任务
      - query:         查询任务状态
    """
    op = request.operation
    task_id = request.task_id

    try:
        if op == 'force_trigger':
            if not request.src_bay or not request.dst_bay:
                response.success = False
                response.message = "force_trigger 需要 src_bay 和 dst_bay"
                return response

            new_id = f"MANUAL_{int(_time_module.time())}"
            ctx = {
                'task_id': new_id,
                'src_bay': request.src_bay,
                'dst_bay': request.dst_bay,
                'robot_id': request.robot_id or 0,
                'cargo_type': request.cargo_type or 1,
            }

            future = asyncio.run_coroutine_threadsafe(
                self._engine.create_instance('agv_transport_atomic', ctx),
                self._loop
            )

            def on_created(fut):
                try:
                    result = fut.result()
                    self.logger.info(f"手动任务下发完成: {new_id}, result={result}")
                except Exception as e:
                    self.logger.error(f"手动任务下发失败: {new_id} {e}")

            future.add_done_callback(on_created)

            response.success = True
            response.message = f"强制任务已下发"
            response.task_id_out = new_id

            # 记录异常日志
            self._engine._persistence.log_exception(
                new_id, 'force_trigger',
                f"src={request.src_bay} dst={request.dst_bay} robot={request.robot_id}"
            )

        elif op == 'cancel':
            if not task_id:
                response.success = False
                response.message = "cancel 需要 task_id"
                return response

            instance = self._engine.get_instance(task_id)
            if not instance:
                response.success = False
                response.message = f"任务不存在: {task_id}"
                return response

            future = asyncio.run_coroutine_threadsafe(
                self._engine.cancel_instance(task_id),
                self._loop
            )
            response.success = True
            response.message = f"任务取消命令已发送: {task_id}"
            response.task_status = 'cancelling'
            response.task_id_out = task_id

            # 记录异常日志
            self._engine._persistence.log_exception(
                task_id, 'user_cancel',
                f"state_before={instance.current_state}"
            )

        elif op == 'retry':
            if not task_id:
                response.success = False
                response.message = "retry 需要 task_id"
                return response

            instance = self._engine.get_instance(task_id)
            if not instance:
                response.success = False
                response.message = f"任务不存在: {task_id}"
                return response

            if instance.current_state != 'manual_intervention':
                response.success = False
                response.message = (
                    f"任务不在 manual_intervention 状态 "
                    f"(当前: {instance.current_state}), 无法重试"
                )
                return response

            future = asyncio.run_coroutine_threadsafe(
                self._engine.inject_event(task_id, 'user_retry', {}),
                self._loop
            )
            response.success = True
            response.message = f"任务重试命令已发送: {task_id}"
            response.task_status = 'retrying'
            response.task_id_out = task_id

            # 记录异常日志
            self._engine._persistence.log_exception(
                task_id, 'user_retry',
                f"state_before={instance.current_state}"
            )

        elif op == 'query':
            if not task_id:
                response.success = False
                response.message = "query 需要 task_id"
                return response

            instance = self._engine.get_instance(task_id)
            if not instance:
                # 尝试从 DB 查询
                db_row = self._engine._persistence.get_instance(task_id)
                if db_row:
                    response.success = True
                    response.message = "任务已归档 (DB)"
                    response.task_status = db_row.get('status', 'UNKNOWN')
                    response.current_state = db_row.get('current_state', '')
                    response.task_id_out = task_id
                else:
                    response.success = False
                    response.message = f"任务不存在: {task_id}"
                return response

            response.success = True
            response.message = f"任务活跃中"
            response.task_status = instance.status
            response.current_state = instance.current_state
            response.task_id_out = task_id

        else:
            response.success = False
            response.message = (
                f"不支持的操作: '{op}' "
                f"(支持: force_trigger, cancel, retry, query)"
            )

    except Exception as e:
        response.success = False
        response.message = f"操作异常: {e}"
        self.logger.error(f"/task_control 异常 [{op}] {task_id}: {e}")

    return response

# =======================================================
```

#### 3.1.3 Action 超时时记录

**文件路径**: `app_mgr_object/components/workflow/instance.py` **(修改)**

在 `_execute_with_timeout` 方法的 `asyncio.TimeoutError` 分支中:

```python
async def _execute_with_timeout(self, state_def) -> str:
    max_retries = state_def.retry or 0
    timeout = state_def.timeout or 0

    for attempt in range(max_retries + 1):
        try:
            if timeout > 0:
                event = await asyncio.wait_for(
                    self._execute_action(state_def.action),
                    timeout=timeout
                )
            else:
                event = await self._execute_action(state_def.action)
            return event

        except asyncio.TimeoutError:
            self._log_warning(f"Action 超时: {state_def.action} "
                              f"(attempt {attempt+1}/{max_retries+1})")
            # === 新增: 记录超时异常 ===
            if attempt >= max_retries:
                try:
                    import json
                    self._persistence.log_exception(
                        self.id, 'action_timeout',
                        json.dumps({
                            'action': state_def.action,
                            'state': self.current_state,
                            'timeout': timeout,
                            'max_retries': max_retries,
                        })
                    )
                except Exception:
                    pass
            # ============================
            if attempt >= max_retries:
                return 'timeout'

        except Exception as e:
            self._log_error(f"Action 失败: {state_def.action} {e}")
            if attempt >= max_retries:
                return 'action_error'

    return 'max_retry'
```

```python
# ====== 新增: 异常日志查询接口 (供 Web 面板 / 运维 CLI 调用) ======

def query_alarms_by_task(self, task_id: str, limit: int = 50) -> list:
    """查询指定任务的告警日志"""
    return self._engine._persistence.query_alarms_by_task(task_id, limit)

def query_exceptions_by_task(self, task_id: str, limit: int = 50) -> list:
    """查询指定任务的异常事件日志"""
    return self._engine._persistence.query_exceptions_by_task(task_id, limit)

def query_state_changes(self, task_id: str) -> list:
    """查询指定任务的状态变更历史"""
    return self._engine._persistence.query_state_changes(task_id)

def get_exception_summary(self) -> dict:
    """获取异常日志汇总统计 (供 Web 仪表盘)"""
    try:
        db_path = self._engine._persistence._db_path
        import sqlite3
        with sqlite3.connect(db_path) as conn:
            conn.row_factory = sqlite3.Row

            # 各事件类型计数
            cursor = conn.execute(
                "SELECT event_type, COUNT(*) as cnt FROM exception_log "
                "GROUP BY event_type ORDER BY cnt DESC"
            )
            by_type = {row['event_type']: row['cnt'] for row in cursor.fetchall()}

            # 最近 24 小时异常数
            cursor = conn.execute(
                "SELECT COUNT(*) as cnt FROM exception_log "
                "WHERE logged_at > datetime('now', '-1 day')"
            )
            recent_24h = cursor.fetchone()['cnt']

            # 告警按等级分布
            cursor = conn.execute(
                "SELECT alarm_level, COUNT(*) as cnt FROM alarm_logs "
                "GROUP BY alarm_level"
            )
            by_alarm_level = {row['alarm_level']: row['cnt']
                              for row in cursor.fetchall()}

            return {
                'total_exceptions': sum(by_type.values()),
                'by_event_type': by_type,
                'recent_24h': recent_24h,
                'alarms': by_alarm_level,
                'total_alarms': sum(by_alarm_level.values()),
            }
    except Exception:
        return {'error': '无法获取日志统计'}

# ================================================================
```
