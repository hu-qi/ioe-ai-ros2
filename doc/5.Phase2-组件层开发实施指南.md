### A.1 `rcs_http_client.py`（核心）

**路径**: `app_mgr_object/components/rcs_client/rcs_http_client.py`

```python
#!/usr/bin/env python3
"""
RCS HTTP 客户端 — 封装 RCS-2000 V3.3 全部对外任务接口

依赖：
  - NetworkPlugin（通过 node.network_plugin 获取 HTTP 会话）
  - 无 ROS2 依赖，纯 Python 组件

RCS-2000 V3.3 接口映射：
  2.1.1  genAgvSchedulingTask   → gen_agv_scheduling_task()
  2.1.2  continueTask           → continue_task()
  2.1.3  cancelTask             → cancel_task()
  2.2.1  agvCallback            → （由 rcs_callback_receiver_plugin 通过 HTTP 回调处理）
  2.2.2  warnCallback           → （同上）
  3.1.6  queryPodBerthAndMat    → query_pod_berth_and_mat()
  3.1.8  bindCtnrAndBin         → bind_ctnr_and_bin()
  3.1.9  queryTaskStatus        → query_task_status()
  3.1.10 queryAgvStatus         → query_agv_status()
"""

import time
import logging
from typing import Dict, Any, List, Optional


class RCSClientError(Exception):
    """RCS 客户端异常"""
    pass


class RCSHttpClient:
    """RCS-2000 V3.3 HTTP 接口客户端

    依赖 node.network_plugin 提供的 HTTP 会话能力（接口绑定 + IP 获取）。
    所有方法均为同步调用，由调用方（插件）决定是否放入线程池。

    使用方式：
        client = RCSHttpClient(
            network_plugin=node.network_plugin,
            base_url="http://192.168.1.100:8080",
            timeout=10.0,
            max_retries=2,
            logger=node.get_logger()
        )
        resp = client.query_agv_status(robot_ids=[1001, 1002])
    """

    def __init__(self,
                 network_plugin,               # NetworkPlugin 实例
                 base_url: str,
                 timeout: float = 10.0,
                 max_retries: int = 2,
                 logger: Optional[logging.Logger] = None):
        """
        Args:
            network_plugin: NetworkPlugin 实例，提供 request() 方法
            base_url: RCS 服务端地址，如 "http://192.168.1.100:8080"
            timeout: 单次请求超时（秒）
            max_retries: 最大重试次数（不含首次）
            logger: 日志记录器
        """
        self._network = network_plugin
        self._base_url = base_url.rstrip('/')
        self._timeout = timeout
        self._max_retries = max_retries
        self._logger = logger or logging.getLogger(__name__)

    # ═══════════════════════════════════════════════════════════════
    # 任务接口 (2.x)
    # ═══════════════════════════════════════════════════════════════

    def gen_agv_scheduling_task(self, payload: Dict[str, Any]) -> Dict[str, Any]:
        """2.1.1 生产任务单

        Args:
            payload: 请求体，由 RCSPayloadBuilder.build_task_payload() 构造
                {
                    "taskCode": "TASK_1723456789_01",
                    "taskTyp": "F13",
                    "priority": 64,
                    "agvCode": "1001",
                    "positionCodePath": [
                        {"positionCode": "A1001", "type": "00"},
                        {"positionCode": "T2001", "type": "00"}
                    ]
                }

        Returns:
            RCS 响应体，含 taskNo（任务流水号）
        """
        return self._post('/genAgvSchedulingTask', payload)

    def continue_task(self, task_code: str) -> Dict[str, Any]:
        """2.1.2 继续执行任务

        Args:
            task_code: 任务号（RCS 返回的 taskNo）
        """
        return self._post('/continueTask', {'taskCode': task_code})

    def cancel_task(self, task_code: str, force_cancel: str = "1") -> Dict[str, Any]:
        """2.1.3 取消任务

        Args:
            task_code: 任务号
            force_cancel: "1"=软取消（叉车背货回库）, "0"=硬取消（就地丢弃）

        注意：默认使用软取消 "1"，避免货物滞留通道造成安全隐患。
        """
        return self._post('/cancelTask', {
            'taskCode': task_code,
            'forceCancel': force_cancel
        })

    # ═══════════════════════════════════════════════════════════════
    # 查询接口 (3.x)
    # ═══════════════════════════════════════════════════════════════

    def query_agv_status(self, robot_ids: List[int]) -> Dict[str, Any]:
        """3.1.10 查询 AGV 状态

        Args:
            robot_ids: AGV 编号列表，如 [1001, 1002]

        Returns:
            RCS 响应体，含每台 AGV 的状态码（1=IDLE, 2=BUSY, 3=OFFLINE, 4=ERROR）
        """
        return self._post('/queryAgvStatus', {'robotIDs': robot_ids})

    def query_task_status(self, task_codes: List[str]) -> Dict[str, Any]:
        """3.1.9 查询任务状态

        Args:
            task_codes: RCS 任务号列表

        Returns:
            RCS 响应体，含每个任务的状态（9=已结束, 其他=执行中/异常）
        """
        return self._post('/queryTaskStatus', {'taskCodes': task_codes})

    def query_pod_berth_and_mat(self, stg_bin_codes: List[str]) -> Dict[str, Any]:
        """3.1.6 查询货架储位与物料批次关系

        用于查询终点仓位（T2001~T4006）的绑定状态，判断是否为空。

        Args:
            stg_bin_codes: 仓位编码列表

        Returns:
            RCS 响应体，含每个仓位的绑定关系数据。
            当 ctnrCode 为空时，表示该仓位无货（空）。
        """
        return self._post('/queryPodBerthAndMat', {'stgBinCodes': stg_bin_codes})

    # ═══════════════════════════════════════════════════════════════
    # 绑定接口 (3.x)
    # ═══════════════════════════════════════════════════════════════

    def bind_ctnr_and_bin(self,
                          ctnr_code: str,
                          ctnr_typ: str,
                          stg_bin_code: str,
                          ind_bind: str = "1") -> Dict[str, Any]:
        """3.1.8 容器与仓位绑定/解绑

        在起始仓位 AI 检测确认有货后，调用此接口绑定容器与仓位。
        绑定后该仓位才能被调度系统识别为"可调度"。

        Args:
            ctnr_code:   容器编码（通常等于仓位编号）
            ctnr_typ:    容器类型（货物类型编号字符串，如 "1"）
            stg_bin_code: 仓位编码
            ind_bind:    "1"=绑定, "0"=解绑
        """
        return self._post('/bindCtnrAndBin', {
            'ctnrCode': ctnr_code,
            'ctnrTyp': ctnr_typ,
            'stgBinCode': stg_bin_code,
            'indBind': ind_bind
        })

    # ═══════════════════════════════════════════════════════════════
    # 内部方法
    # ═══════════════════════════════════════════════════════════════

    def _post(self, path: str, data: Dict[str, Any]) -> Dict[str, Any]:
        """发送 POST 请求，含自动重试

        Args:
            path: 接口路径（如 '/queryAgvStatus'）
            data: JSON 请求体

        Returns:
            解析后的 JSON 响应体

        Raises:
            RCSClientError: 所有重试均失败时抛出
        """
        url = f"{self._base_url}{path}"
        last_error = None

        for attempt in range(self._max_retries + 1):
            try:
                self._log_debug(f"RCS 请求 [{attempt+1}/{self._max_retries+1}]: POST {url}")
                resp = self._network.request(
                    'POST',
                    url,
                    data=data,
                    timeout=self._timeout
                )
                result = resp.json()
                self._log_debug(f"RCS 响应: {url} → code={result.get('code')}")
                return result

            except Exception as e:
                last_error = e
                if attempt < self._max_retries:
                    wait = 0.5 * (attempt + 1)
                    self._log_warning(f"RCS 请求失败 [{attempt+1}], {wait}s 后重试: {e}")
                    time.sleep(wait)

        raise RCSClientError(f"RCS 请求失败 [{path}] 已重试 {self._max_retries} 次: {last_error}")

    def _log_debug(self, msg: str):
        if self._logger:
            self._logger.debug(msg)

    def _log_warning(self, msg: str):
        if self._logger:
            self._logger.warning(msg)
```

```python
#!/usr/bin/env python3
"""
RCS 请求体构造器 — 标准化 genAgvSchedulingTask 接口的请求 payload

使用方式：
    builder = RCSPayloadBuilder()
    payload = builder.build_task_payload(
        task_context={
            'task_id': 'TASK_1723456789_01',
            'src_bay': 'A1001',
            'dst_bay': 'T2001',
            'robot_id': 1001,
        },
        task_typ='F13',
        priority=64
    )
"""

from typing import Dict, Any


class RCSPayloadBuilder:
    """构造 RCS-2000 V3.3 接口标准请求体"""

    @staticmethod
    def build_task_payload(task_context: Dict[str, Any],
                           task_typ: str = "F13",
                           priority: int = 64) -> Dict[str, Any]:
        """构造 genAgvSchedulingTask 请求体

        Args:
            task_context: 任务上下文，必须包含:
                - task_id:   任务唯一标识
                - src_bay:   起始仓位编号（如 "A1001"）
                - dst_bay:   终点仓位编号（如 "T2001"）
                - robot_id:  分配的 AGV 编号（如 1001）
            task_typ: 任务类型码，默认 "F13"（巷道到工作台）
            priority: 优先级 1~127，值越大越高，默认 64

        Returns:
            符合 RCS 接口规范的 JSON 请求体
        """
        return {
            'taskCode': task_context['task_id'],
            'taskTyp': task_typ,
            'priority': priority,
            'agvCode': str(task_context['robot_id']),
            'positionCodePath': [
                {
                    'positionCode': task_context['src_bay'],
                    'type': '00'
                },
                {
                    'positionCode': task_context['dst_bay'],
                    'type': '00'
                }
            ]
        }

    @staticmethod
    def build_bind_payload(bay_id: str, cargo_type: int,
                           ind_bind: str = "1") -> Dict[str, Any]:
        """构造 bindCtnrAndBin 请求体

        Args:
            bay_id:     仓位编号
            cargo_type: 货物类型 1~6
            ind_bind:   "1"=绑定, "0"=解绑

        Returns:
            绑定请求体
        """
        return {
            'ctnrCode': bay_id,
            'ctnrTyp': str(cargo_type),
            'stgBinCode': bay_id,
            'indBind': ind_bind
        }

    @staticmethod
    def build_force_dispatch_payload(src_bay: str, dst_bay: str,
                                     robot_id: int, task_id: str,
                                     task_typ: str = "F13",
                                     priority: int = 64) -> Dict[str, Any]:
        """构造强制下发（手动）任务的请求体

        与自动任务的区别仅在于 task_id 的前缀（FORCE_ 而非 TASK_）。
        """
        return RCSPayloadBuilder.build_task_payload(
            task_context={
                'task_id': task_id,
                'src_bay': src_bay,
                'dst_bay': dst_bay,
                'robot_id': robot_id,
            },
            task_typ=task_typ,
            priority=priority
        )
```

### A.3 `rcs_response_parser.py`

**路径**: `app_mgr_object/components/rcs_client/rcs_response_parser.py`

```python
#!/usr/bin/env python3
"""
RCS 响应解析器 — 将 RCS HTTP 响应体解析为组件层标准数据结构

设计原则：
  - 响应解析与 HTTP 调用分离，便于单元测试
  - 对 RCS 返回的各种 code 做统一解释
  - 解析失败时返回安全的默认值而非抛异常
"""

from typing import Dict, Any, List, Optional


# ═══════════════════════════════════════════════════════════════
# RCS 状态码常量
# ═══════════════════════════════════════════════════════════════

class RCSRobotStatus:
    """AGV 状态码"""
    IDLE    = "1"   # 空闲
    BUSY    = "2"   # 执行中
    OFFLINE = "3"   # 离线
    ERROR   = "4"   # 异常


class RCSTaskStatus:
    """任务状态码"""
    PENDING     = "0"   # 待执行
    DISPATCHING = "1"   # 下发中
    RUNNING     = "2"   # 执行中
    PAUSED      = "3"   # 暂停
    COMPLETED   = "9"   # 已结束（正常完成）
    CANCELLED   = "8"   # 已取消
    FAILED      = "7"   # 失败


class RCSResponseCode:
    """RCS 通用响应 code"""
    SUCCESS = "0"   # 成功


class RCSResponseParser:
    """RCS 响应解析器 — 提供静态方法，不维护状态"""

    @staticmethod
    def is_success(resp: Dict[str, Any]) -> bool:
        """判断 RCS 响应是否成功"""
        return str(resp.get('code', '')) == RCSResponseCode.SUCCESS

    @staticmethod
    def parse_agv_status(resp: Dict[str, Any]) -> Dict[int, str]:
        """解析 queryAgvStatus 响应 → {robot_id: status}

        Returns:
            {1001: 'IDLE', 1002: 'BUSY'} 等
            解析失败返回空 dict
        """
        result = {}
        try:
            data_list = resp.get('data', [])
            if not isinstance(data_list, list):
                return result
            status_map = {
                RCSRobotStatus.IDLE:    'IDLE',
                RCSRobotStatus.BUSY:    'BUSY',
                RCSRobotStatus.OFFLINE: 'OFFLINE',
                RCSRobotStatus.ERROR:   'ERROR',
            }
            for item in data_list:
                robot_id = item.get('robotID') or item.get('agvCode')
                if robot_id is None:
                    continue
                status_code = str(item.get('status', item.get('agvStatus', '')))
                result[int(robot_id)] = status_map.get(status_code, 'UNKNOWN')
        except Exception:
            pass
        return result

    @staticmethod
    def parse_task_status(resp: Dict[str, Any]) -> Dict[str, str]:
        """解析 queryTaskStatus 响应 → {task_code: status}

        Returns:
            {'TASK_xxx': 'COMPLETED', ...}
            解析失败返回空 dict
        """
        result = {}
        try:
            data_list = resp.get('data', [])
            if not isinstance(data_list, list):
                return result
            status_map = {
                RCSTaskStatus.PENDING:     'PENDING',
                RCSTaskStatus.DISPATCHING: 'DISPATCHING',
                RCSTaskStatus.RUNNING:     'RUNNING',
                RCSTaskStatus.PAUSED:      'PAUSED',
                RCSTaskStatus.COMPLETED:   'COMPLETED',
                RCSTaskStatus.CANCELLED:   'CANCELLED',
                RCSTaskStatus.FAILED:      'FAILED',
            }
            for item in data_list:
                task_code = item.get('taskCode', '')
                if not task_code:
                    continue
                status_code = str(item.get('taskStatus', ''))
                result[task_code] = status_map.get(status_code, 'UNKNOWN')
        except Exception:
            pass
        return result

    @staticmethod
    def parse_gen_task_response(resp: Dict[str, Any]) -> Optional[str]:
        """解析 genAgvSchedulingTask 响应 → 返回 taskNo（RCS 任务流水号）

        Returns:
            成功时返回 RCS 任务流水号，失败返回 None
        """
        try:
            if RCSResponseParser.is_success(resp):
                return resp.get('data', {}).get('taskNo', '')
        except Exception:
            pass
        return None

    @staticmethod
    def parse_pod_berth(resp: Dict[str, Any]) -> Dict[str, bool]:
        """解析 queryPodBerthAndMat 响应 → {bay_id: is_empty}

        is_empty=True 表示该仓位为空（无容器绑定）
        """
        result = {}
        try:
            data_list = resp.get('data', [])
            if not isinstance(data_list, list):
                return result
            for item in data_list:
                bay_id = item.get('stgBinCode', '')
                if not bay_id:
                    continue
                ctnr_code = item.get('ctnrCode', '')
                result[bay_id] = (ctnr_code is None or ctnr_code == '')
        except Exception:
            pass
        return result

    @staticmethod
    def get_task_no(resp: Dict[str, Any]) -> str:
        """从 genAgvSchedulingTask 响应中提取 RCS 任务流水号

        Returns:
            任务流水号字符串，失败返回空字符串
        """
        return RCSResponseParser.parse_gen_task_response(resp) or ''
```

```python
#!/usr/bin/env python3
"""
RCS HTTP 客户端组件包

提供：
  - RCSHttpClient        : RCS-2000 V3.3 HTTP 接口封装
  - RCSPayloadBuilder     : 请求体构造器
  - RCSResponseParser     : 响应解析器

  - RCSTaskStatus         : 任务状态码常量
```

```python
#!/usr/bin/env python3
"""
起始仓位状态缓存 — 维护 24 个起始仓位（A1001~D1006）的实时状态

数据结构（内存）:
    _bays: {
        "A1001": {
            "bay_id":       "A1001",
            "cargo_type":   1,
            "ai_detected":  True,
            "bind_status":  1,     # 0=未绑定, 1=已绑定, 2=绑定中
            "debounce_count": 3,
            "in_task":      False, # 是否已被分配任务
            "last_seen":    1723456789.0,
            "bind_time":    1723456790.0,
        },
        ...
    }
"""

import time
import threading
from typing import Dict, List, Optional


class BayCacheEntry:
    """单个仓位状态条目"""
    __slots__ = ('bay_id', 'cargo_type', 'ai_detected', 'bind_status',
                 'debounce_count', 'in_task', 'last_seen', 'bind_time')

    def __init__(self, bay_id: str, cargo_type: int):
        self.bay_id = bay_id
        self.cargo_type = cargo_type
        self.ai_detected = False
        self.bind_status = 0
        self.debounce_count = 0
        self.in_task = False
        self.last_seen = 0.0
        self.bind_time = 0.0

    def to_dict(self) -> dict:
        return {
            'bay_id': self.bay_id,
            'cargo_type': self.cargo_type,
            'ai_detected': self.ai_detected,
            'bind_status': self.bind_status,
            'debounce_count': self.debounce_count,
            'in_task': self.in_task,
            'last_seen': self.last_seen,
            'bind_time': self.bind_time,
        }


class BayCache:
    """起始仓位状态缓存"""

    def __init__(self, bay_configs: Dict[str, int] = None, logger=None):
        """
        Args:
            bay_configs: {bay_id: cargo_type} 初始配置
            logger: 日志记录器
        """
        self._lock = threading.RLock()
        self._logger = logger
        self._bays: Dict[str, BayCacheEntry] = {}

        if bay_configs:
            for bay_id, cargo_type in bay_configs.items():
                self._bays[bay_id] = BayCacheEntry(bay_id, cargo_type)

    # ── 写入接口 ──

    def update_ai_result(self, bay_id: str, has_cargo: bool, cargo_type: int = None):
        """更新 AI 检测原始结果"""
        with self._lock:
            entry = self._ensure_entry(bay_id, cargo_type)
            entry.ai_detected = has_cargo
            entry.last_seen = time.time()

    def increment_debounce(self, bay_id: str):
        """递增防抖计数"""
        with self._lock:
            entry = self._ensure_entry(bay_id, None)
            entry.debounce_count += 1

    def reset_debounce(self, bay_id: str):
        """重置防抖计数（收到无货信号时调用）"""
        with self._lock:
            entry = self._ensure_entry(bay_id, None)
            entry.debounce_count = 0

    def set_bound(self, bay_id: str):
        """标记仓位已绑定"""
        with self._lock:
            entry = self._ensure_entry(bay_id, None)
            entry.bind_status = 1
            entry.bind_time = time.time()

    def set_binding(self, bay_id: str):
        """标记仓位绑定中"""
        with self._lock:
            entry = self._ensure_entry(bay_id, None)
            entry.bind_status = 2

    def set_unbound(self, bay_id: str):
        """标记仓位未绑定"""
        with self._lock:
            entry = self._ensure_entry(bay_id, None)
            entry.bind_status = 0
            entry.bind_time = 0.0

    def mark_in_task(self, bay_id: str, value: bool = True):
        """标记仓位被任务占用 / 释放"""
        with self._lock:
            entry = self._ensure_entry(bay_id, None)
            entry.in_task = value

    def update_cargo_type(self, bay_id: str, cargo_type: int):
        """更新仓位货物类型（配置热更新）"""
        with self._lock:
            if bay_id in self._bays:
                self._bays[bay_id].cargo_type = cargo_type

    # ── 查询接口 ──

    def get_bound_bays_of_type(self, cargo_type: int,
                               exclude_in_task: bool = True) -> List[str]:
        """获取指定类型下已绑定且可调度的仓位列表"""
        with self._lock:
            result = []
            for entry in self._bays.values():
                if entry.cargo_type != cargo_type:
                    continue
                if entry.bind_status != 1:
                    continue
                if exclude_in_task and entry.in_task:
                    continue
                result.append(entry.bay_id)
            return result

    def get_any_bound_bay(self, cargo_type: int) -> Optional[str]:
        """获取指定类型下任一已绑定且未分配任务的仓位"""
        bays = self.get_bound_bays_of_type(cargo_type, exclude_in_task=True)
        return bays[0] if bays else None

    def get_bay_status(self, bay_id: str) -> Optional[dict]:
        """获取单个仓位完整状态"""
        with self._lock:
            entry = self._bays.get(bay_id)
            return entry.to_dict() if entry else None

    def get_all_bay_status(self) -> Dict[str, dict]:
        """获取所有仓位状态快照"""
        with self._lock:
            return {bay_id: entry.to_dict()
                    for bay_id, entry in self._bays.items()}

    def get_bay_count(self) -> int:
        """获取仓位总数"""
        return len(self._bays)

    def get_bound_count(self) -> int:
        """获取已绑定仓位数量"""
        with self._lock:
            return sum(1 for e in self._bays.values() if e.bind_status == 1)

    def has_bound_bay_of_type(self, cargo_type: int) -> bool:
        """指定类型是否有已绑定且未分配任务的仓位"""
        return len(self.get_bound_bays_of_type(cargo_type)) > 0

    # ── 内部方法 ──

    def _ensure_entry(self, bay_id: str,
                      cargo_type: Optional[int]) -> BayCacheEntry:
        """确保仓位条目存在，不存在则创建"""
        if bay_id not in self._bays:
            ct = cargo_type if cargo_type is not None else 0
            self._bays[bay_id] = BayCacheEntry(bay_id, ct)
            if self._logger:
                self._logger.debug(f"新仓位注册: {bay_id} type={ct}")
        elif cargo_type is not None and cargo_type > 0:
            current = self._bays[bay_id]
            if current.cargo_type == 0:
                current.cargo_type = cargo_type
        return self._bays[bay_id]

    def reload_config(self, bay_configs: Dict[str, int]):
        """热加载仓位配置（用于 Web 配置更新后）"""
        with self._lock:
            for bay_id, cargo_type in bay_configs.items():
                if bay_id in self._bays:
                    self._bays[bay_id].cargo_type = cargo_type
                else:
                    self._bays[bay_id] = BayCacheEntry(bay_id, cargo_type)
```

```python
        """
        Args:
            robot_ids: AGV 编号列表，如 [1001, 1002]
            logger: 日志记录器

        self._logger = logger

        if robot_ids:

            entry = self._robots.get(robot_id)
            if not entry:
                return
            entry.status = status
            entry.update_time = time.time()
            if battery is not None:
                entry.battery = battery
            if position_code is not None:
                entry.position_code = position_code

        with self._lock:
            entry = self._robots.get(robot_id)
            if entry:
                entry.status = 'BUSY'
                entry.current_task_id = task_id
                entry.update_time = time.time()

    def set_idle(self, robot_id: int):
        """标记叉车为空闲"""
        with self._lock:
            entry = self._robots.get(robot_id)
            if entry:
                entry.status = 'IDLE'
                entry.current_task_id = ''
                entry.update_time = time.time()
```

```python
"""

from typing import Dict, List, Any, Optional


class ScheduleEntry:
    """工作时间窗口条目"""
    def __init__(self, day_of_week: str, start_time: str, end_time: str):
        self.day_of_week = day_of_week
        self.start_time = start_time      # "08:00"

class ConfigCache:
    """配置热缓存"""

    def __init__(self, sys_configs: Dict[str, str] = None, logger=None):
        """
        Args:
            sys_configs: 初始系统配置 {key: value}
            logger: 日志记录器
        """
        self._lock = threading.RLock()
        self._logger = logger

        # 系统配置
        self._sys_config: Dict[str, str] = sys_configs or {}

        # 仓位映射
        self._source_bay_mapping: Dict[str, int] = {}   # bay_id → cargo_type
        self._dest_bay_mapping: Dict[str, dict] = {}    # bay_id → {cargo_type, floor}

        # 工作时段
        self._schedules: List[ScheduleEntry] = []

    def get_sys(self, key: str, default: Any = None) -> Any:

        """获取整数类型系统配置"""
        try:
            return int(self.get_sys(key, str(default)))

            return default

    def get_float(self, key: str, default: float = 0.0) -> float:
        """获取浮点类型系统配置"""
        try:
            return float(self.get_sys(key, str(default)))
        except (ValueError, TypeError):
            return default

    def set_sys(self, key: str, value: str):
        """更新系统配置"""
        with self._lock:
            self._sys_config[key] = value
            self._update_time = time.time()

    # ── 仓位映射 ──

    def set_source_mapping(self, mapping: Dict[str, int]):

            self._source_bay_mapping = mapping.copy()
            self._update_time = time.time()

            self._dest_bay_mapping = mapping.copy()
            self._update_time = time.time()

    def get_source_mapping(self) -> Dict[str, int]:
        with self._lock:
            return self._source_bay_mapping.copy()

    def get_dest_mapping(self) -> Dict[str, dict]:
```

```python
#!/usr/bin/env python3
"""
工作流定义加载器 — 从 YAML 文件加载工作流定义并校验

YAML 结构规范：
    name: "agv_transport_atomic"
    version: 1
    initial_state: "calling_rcs"
    states:
      - id: calling_rcs
        action: invoke_gen_task
        retry: 2
        timeout: 10
        transitions:
          - event: "rcs_ack_ok"
            target: "monitoring"
    on_error:
      - action: log_error
"""

import os
import yaml
from typing import Dict, List, Any, Optional


class WorkflowStateDef:
    """工作流状态定义"""
    def __init__(self, state_id: str, raw: Dict[str, Any]):
        self.id = state_id
        self.description = raw.get('description', '')
        self.action = raw.get('action')                     # 主动作函数名
        self.on_enter = raw.get('on_enter', [])             # 进入状态时执行的动作
        self.on_exit = raw.get('on_exit', [])               # 离开状态时执行的动作
        self.retry = raw.get('retry', 0)                    # 重试次数
        self.timeout = raw.get('timeout', 0)                # 超时时间（秒）
        self.final = raw.get('final', False)                # 是否为终态
        self.transitions: Dict[str, str] = {}               # event → target_state

        # 解析转移
        for trans in raw.get('transitions', []):
            event = trans.get('event', '')
            target = trans.get('target', '')
            if event and target:
                self.transitions[event] = target


class WorkflowDefinition:
    """工作流定义"""
    def __init__(self, raw: Dict[str, Any]):
        self.name = raw.get('name', 'unnamed')
        self.version = raw.get('version', 1)
        self.initial_state = raw.get('initial_state', '')
        self.max_concurrent = raw.get('max_concurrent', 10)
        self.description = raw.get('description', '')

        # 解析状态
        self.states: Dict[str, WorkflowStateDef] = {}
        for state_raw in raw.get('states', []):
            sid = state_raw.get('id', '')
            if sid:
                self.states[sid] = WorkflowStateDef(sid, state_raw)

        # 全局异常处理
        self.on_error = raw.get('on_error', [])

    def get_state(self, state_id: str) -> Optional[WorkflowStateDef]:
        return self.states.get(state_id)

    def is_final_state(self, state_id: str) -> bool:
        state = self.states.get(state_id)
        return state.final if state else True

    def get_next_state(self, current_state: str, event: str) -> Optional[str]:
        """根据事件获取下一个状态"""
        state = self.states.get(current_state)
        if not state:
            return None
        return state.transitions.get(event)


class WorkflowDefinitionLoader:
    """工作流定义加载器"""

    def __init__(self, definitions_dir: str, logger=None):
        """
        Args:
            definitions_dir: YAML 工作流定义目录
            logger: 日志记录器
        """
        self._dir = definitions_dir
        self._logger = logger

    def load_all(self) -> Dict[str, WorkflowDefinition]:
        """加载所有 YAML 工作流定义"""
        definitions = {}
        if not os.path.isdir(self._dir):
            if self._logger:
                self._logger.warning(f"工作流定义目录不存在: {self._dir}")
            return definitions

        for filename in os.listdir(self._dir):
            if not filename.endswith(('.yaml', '.yml')):
                continue
            filepath = os.path.join(self._dir, filename)
            try:
                with open(filepath, 'r', encoding='utf-8') as f:
                    raw = yaml.safe_load(f)
                if not raw or 'name' not in raw:
                    continue
                definition = WorkflowDefinition(raw)
                definitions[definition.name] = definition
                if self._logger:
                    self._logger.info(f"加载工作流定义: {definition.name} "
                                      f"(v{definition.version}, "
                                      f"{len(definition.states)} 状态)")
            except Exception as e:
                if self._logger:
                    self._logger.error(f"加载工作流定义失败 {filepath}: {e}")

        return definitions

    def load_one(self, name: str) -> Optional[WorkflowDefinition]:
        """加载指定名称的工作流定义"""
        return self.load_all().get(name)
```

```python
Action 函数签名:

import logging
from typing import Dict, Callable, Any


# Action 函数类型标注
ActionFunc = Callable[..., Any]


class ActionRegistry:
    """Action 函数注册表"""

    def __init__(self, logger: logging.Logger = None):
        self._actions: Dict[str, ActionFunc] = {}
        self._logger = logger or logging.getLogger(__name__)

    def register(self, name: str, func: ActionFunc):
        """注册一个 Action 函数"""
        if name in self._actions:
            if self._logger:
                self._logger.warning(f"Action 已存在，覆盖: {name}")
        self._actions[name] = func
        if self._logger:
            self._logger.debug(f"注册 Action: {name}")

    def register_batch(self, actions: Dict[str, ActionFunc]):
        """批量注册 Action 函数"""
        for name, func in actions.items():
            self.register(name, func)

    def get(self, name: str) -> ActionFunc:
        """获取 Action 函数

        Raises:
            KeyError: Action 未注册
        """
        if name not in self._actions:
            raise KeyError(f"Action 未注册: {name}")
        return self._actions[name]

    def has(self, name: str) -> bool:
        """检查 Action 是否已注册"""
        return name in self._actions

    def list_actions(self) -> list:
        """列出所有已注册的 Action 名称"""
        return list(self._actions.keys())

    async def execute(self, name: str, context: dict, instance) -> str:
        """执行指定 Action，返回事件名"""
        func = self.get(name)
        try:
            result = await func(context, instance)
            return str(result) if result else 'done'
        except Exception as e:
            if self._logger:
                self._logger.error(f"Action [{name}] 执行异常: {e}")
```

```python
"""
工作流持久化层 — 基于 SQLite 的工作流实例持久化

表结构（在 config/db_schema.sql 中定义）：
    CREATE TABLE task_instances (
        task_id       TEXT PRIMARY KEY,

import threading
import time
import logging
from typing import Dict, List, Any, Optional

    def __init__(self, db_path: str, logger: logging.Logger = None):

        Args:
            db_path: SQLite 数据库文件路径
            logger: 日志记录器
        """

        self._logger = logger or logging.getLogger(__name__)
        self._lock = threading.Lock()
        self._init_db()

    def _init_db(self):
        """初始化数据库（创建表如不存在）"""
        try:
            with sqlite3.connect(self._db_path) as conn:
                conn.execute("PRAGMA journal_mode=WAL")
                conn.execute("""
                    CREATE TABLE IF NOT EXISTS task_instances (
                        task_id       TEXT PRIMARY KEY,
                        workflow_name TEXT,
                        src_bay       TEXT,
                        dst_bay       TEXT,
                        robot_id      INT,

                        current_state TEXT,
                        context_blob  BLOB,
                        created_at    TIMESTAMP DEFAULT (datetime('now','localtime')),
                        completed_at  TIMESTAMP
                    )
                """)
                conn.commit()
        except Exception as e:
            if self._logger:
                self._logger.error(f"初始化数据库失败: {e}")

    # ── 保存 ──

    def save(self, instance) -> bool:
        """保存工作流实例到数据库（UPSERT）"""
        with self._lock:
            try:
                with sqlite3.connect(self._db_path) as conn:
                    context_blob = pickle.dumps(instance.context)
                    conn.execute("""
                        INSERT OR REPLACE INTO task_instances
                        (task_id, workflow_name, src_bay, dst_bay, robot_id,
                         cargo_type, rcs_task_no, status, current_state, context_blob)
                        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
                    """, (
                        instance.id,
                        instance.definition.name,
                        instance.context.get('src_bay', ''),
                        instance.context.get('dst_bay', ''),
                        instance.context.get('robot_id', 0),
                        instance.context.get('cargo_type', 0),
                        instance.context.get('rcs_task_no', ''),
                        self._status_from_state(instance.current_state),
                        instance.current_state,
                        context_blob
                    ))
                    conn.commit()
                return True
            except Exception as e:
                if self._logger:
                    self._logger.error(f"保存实例失败 {instance.id}: {e}")
                return False

    def update_state(self, task_id: str, state: str):
        """更新实例当前状态"""
        with self._lock:
            try:
                status = self._status_from_state(state)
                with sqlite3.connect(self._db_path) as conn:
                    conn.execute("""
                        UPDATE task_instances
                        SET current_state = ?, status = ?, completed_at = ?
                        WHERE task_id = ?
                    """, (
                        state,
                        status,
                        datetime_now() if state in ('completed', 'failed') else None,
                        task_id
                    ))
                    conn.commit()
            except Exception as e:
                if self._logger:
                    self._logger.error(f"更新状态失败 {task_id}: {e}")

    def update_rcs_task_no(self, task_id: str, rcs_task_no: str):
        """更新 RCS 任务流水号"""
        with self._lock:
            try:
                with sqlite3.connect(self._db_path) as conn:
                    conn.execute("""
                        UPDATE task_instances SET rcs_task_no = ? WHERE task_id = ?
                    """, (rcs_task_no, task_id))
                    conn.commit()
            except Exception as e:
                if self._logger:
                    self._logger.error(f"更新 RCS 任务号失败 {task_id}: {e}")

    # ── 查询 ──

    def get_incomplete_instances(self) -> List[Dict[str, Any]]:
        """获取所有未完成的实例（用于断电恢复）"""
        with self._lock:
            try:
                with sqlite3.connect(self._db_path) as conn:
                    conn.row_factory = sqlite3.Row
                    cursor = conn.execute("""
                        SELECT * FROM task_instances
                        WHERE status NOT IN ('SUCCESS', 'FAILED', 'MANUAL')
                    """)
                    return [dict(row) for row in cursor.fetchall()]
            except Exception as e:
                if self._logger:
                    self._logger.error(f"查询未完成实例失败: {e}")
                return []

    def get_instance(self, task_id: str) -> Optional[Dict[str, Any]]:
        """获取单个任务实例"""
        with self._lock:
            try:
                with sqlite3.connect(self._db_path) as conn:
                    conn.row_factory = sqlite3.Row
                    cursor = conn.execute(
                        "SELECT * FROM task_instances WHERE task_id = ?",
                        (task_id,)
                    )
                    row = cursor.fetchone()
                    return dict(row) if row else None
            except Exception as e:
                if self._logger:
                    self._logger.error(f"查询实例失败 {task_id}: {e}")
                return None

    def get_all_instances(self, limit: int = 100) -> List[Dict[str, Any]]:
        """获取所有任务实例"""
        with self._lock:
            try:
                with sqlite3.connect(self._db_path) as conn:
                    conn.row_factory = sqlite3.Row
                    cursor = conn.execute(
                        "SELECT * FROM task_instances ORDER BY created_at DESC LIMIT ?",
                        (limit,)
                    )
                    return [dict(row) for row in cursor.fetchall()]
            except Exception as e:
                if self._logger:
                    self._logger.error(f"查询实例列表失败: {e}")
                return []

    # ── 工具方法 ──

    @staticmethod
    def _status_from_state(state: str) -> str:

            'calling_rcs':         'DISPATCHING',
            'monitoring':          'RUNNING',
            'compensating_query':  'RUNNING',
            'releasing':           'RUNNING',
            'releasing_and_done':  'RUNNING',
            'completed':           'SUCCESS',
            'failed':              'FAILED',
            'manual_intervention': 'MANUAL',
        }
        return mapping.get(state, 'RUNNING')
```

```python
#!/usr/bin/env python3
"""
工作流实例 — 单个工作流的状态机执行器

设计要点：
  - 每个实例对应一次完整的原子搬运任务
  - 通过 asyncio.Queue 接收外部事件（如 RCS 回调）
  - 每个状态步骤支持超时和重试
  - 状态转移后自动持久化
"""

import asyncio
import time
import logging
from typing import Dict, Any, Optional
from .definition import WorkflowDefinition


class WorkflowInstanceStatus:
    """工作流实例状态"""
    RUNNING = 'running'
    COMPLETED = 'completed'
    FAILED = 'failed'
    CANCELLED = 'cancelled'


class WorkflowInstance:
    """工作流实例 — 驱动 YAML 状态机的执行"""

    def __init__(self,
                 instance_id: str,
                 definition: WorkflowDefinition,
                 context: Dict[str, Any],
                 action_registry,
                 persistence,
                 logger: logging.Logger = None):
        """
        Args:
            instance_id:   任务唯一标识（如 "TASK_1723456789_01"）
            definition:    工作流定义
            context:       任务上下文 {src_bay, dst_bay, robot_id, cargo_type, ...}
            action_registry: ActionRegistry 实例
            persistence:   WorkflowPersistence 实例
            logger:        日志记录器
        """
        self.id = instance_id
        self.definition = definition
        self.context = context
        self._action_registry = action_registry
        self._persistence = persistence
        self._logger = logger or logging.getLogger(__name__)

        self.current_state = definition.initial_state
        self._status = WorkflowInstanceStatus.RUNNING
        self._retry_count = 0
        self._start_time = time.time()
        self._event_queue: asyncio.Queue = asyncio.Queue()

        # 回调
        self.on_state_changed = None           # callable(state_id)
        self.on_completed = None               # callable(instance)
        self.on_failed = None                  # callable(instance, error)

    # ═══════════════════════════════════════════════════════════
    # 主执行循环
    # ═══════════════════════════════════════════════════════════

    async def run(self):
        """执行工作流，直到到达终态"""
        self._log_info(f"工作流启动: {self.id} 起始状态={self.current_state}")
        self._persistence.save(self)

        try:
            while not self.definition.is_final_state(self.current_state):
                state_def = self.definition.get_state(self.current_state)
                if not state_def:
                    self._log_error(f"无效状态: {self.current_state}")
                    await self._transition_to('failed')
                    break

                # 1. 执行 on_enter 动作
                for action_name in state_def.on_enter:
                    await self._execute_action(action_name)

                # 2. 执行主动作（带超时控制）
                if state_def.action:
                    event = await self._execute_with_timeout(state_def)
                else:
                    # 无主动作，等待外部事件
                    event = await self._wait_for_event(state_def.timeout)

                # 3. 状态转移
                next_state = state_def.transitions.get(event)
                if next_state:
                    await self._transition_to(next_state)
                else:
                    self._log_warning(f"无匹配转移: state={self.current_state} event={event}")
                    await self._transition_to('manual_intervention')

            # 终态
            self._status = WorkflowInstanceStatus.COMPLETED
            self._log_info(f"工作流完成: {self.id} 终态={self.current_state}")

        except asyncio.CancelledError:
            self._status = WorkflowInstanceStatus.CANCELLED
            self._log_warning(f"工作流取消: {self.id}")

        except Exception as e:
            self._status = WorkflowInstanceStatus.FAILED
            self._log_error(f"工作流异常: {self.id} {e}")

        finally:
            self._persistence.save(self)

    # ═══════════════════════════════════════════════════════════
    # 外部事件注入
    # ═══════════════════════════════════════════════════════════

    async def inject_event(self, event: str, data: Dict[str, Any] = None):
        """注入外部事件（如 RCS 回调到达时）"""
        await self._event_queue.put({'event': event, 'data': data or {}})

    # ═══════════════════════════════════════════════════════════
    # 状态转移
    # ═══════════════════════════════════════════════════════════

    async def _transition_to(self, state_id: str):
        """执行状态转移"""
        old_state = self.current_state

        # 执行 on_exit 动作
        old_def = self.definition.get_state(old_state)
        if old_def:
            for action_name in old_def.on_exit:
                await self._execute_action(action_name)

        self.current_state = state_id
        self._retry_count = 0
        self._log_debug(f"状态转移: {old_state} → {state_id}")

        # 持久化
        self._persistence.save(self)

        # 回调通知
        if self.on_state_changed:
            try:
                self.on_state_changed(state_id)
            except Exception:
                pass

    # ═══════════════════════════════════════════════════════════
    # Action 执行
    # ═══════════════════════════════════════════════════════════

    async def _execute_action(self, action_name: str) -> str:
        """执行单个 Action，返回事件名"""
        event = await self._action_registry.execute(
            action_name, self.context, self
        )
        return event

    async def _execute_with_timeout(self, state_def) -> str:
        """执行带超时和重试的主动作"""
        max_retries = state_def.retry or 0
        timeout = state_def.timeout or 0

        for attempt in range(max_retries + 1):
            try:
                if timeout > 0:
                    event = await asyncio.wait_for(
                        self._execute_action(state_def.action),
                        timeout=timeout
                    )
                else:
                    event = await self._execute_action(state_def.action)
                return event

            except asyncio.TimeoutError:
                self._log_warning(f"Action 超时: {state_def.action} "
                                  f"(attempt {attempt+1}/{max_retries+1})")
                if attempt >= max_retries:
                    return 'timeout'

            except Exception as e:
                self._log_error(f"Action 失败: {state_def.action} {e}")
                if attempt >= max_retries:
                    return 'action_error'

        return 'max_retry'

    async def _wait_for_event(self, timeout: float = 0) -> str:
        """等待外部事件，支持超时"""
        try:
            if timeout > 0:
                msg = await asyncio.wait_for(self._event_queue.get(), timeout=timeout)
            else:
                msg = await self._event_queue.get()
            return msg.get('event', 'unknown')
        except asyncio.TimeoutError:
            return 'timeout'

    # ═══════════════════════════════════════════════════════════
    # 属性
    # ═══════════════════════════════════════════════════════════

    @property
    def status(self) -> str:
        return self._status

    @property
    def is_running(self) -> bool:
        return self._status == WorkflowInstanceStatus.RUNNING

    @property
    def duration(self) -> float:
        return time.time() - self._start_time

    def _log_info(self, msg: str):
        if self._logger:
            self._logger.info(f"[{self.id}] {msg}")

    def _log_warning(self, msg: str):
        if self._logger:
            self._logger.warning(f"[{self.id}] {msg}")

    def _log_error(self, msg: str):
        if self._logger:
            self._logger.error(f"[{self.id}] {msg}")

    def _log_debug(self, msg: str):
        if self._logger:
            self._logger.debug(f"[{self.id}] {msg}")
```

```python
#!/usr/bin/env python3
"""
工作流引擎 — 管理所有工作流实例的生命周期

职责：
  - 创建新实例
  - 恢复未完成实例（断电恢复）
  - 控制并发数
  - 提供外部事件注入入口
"""

import asyncio
import logging
from typing import Dict, Any, Optional, Callable
from .definition import WorkflowDefinitionLoader, WorkflowDefinition
from .instance import WorkflowInstance
from .action_registry import ActionRegistry
from .persistence import WorkflowPersistence


class WorkflowEngine:
    """工作流引擎"""

    def __init__(self,
                 definitions_dir: str,
                 db_path: str,
                 max_concurrent: int = 10,
                 logger: logging.Logger = None):
        """
        Args:
            definitions_dir: YAML 工作流定义目录
            db_path:        SQLite 数据库文件路径
            max_concurrent: 最大并发实例数
            logger:         日志记录器
        """
        self._logger = logger or logging.getLogger(__name__)
        self._max_concurrent = max_concurrent

        # 子模块
        self._def_loader = WorkflowDefinitionLoader(definitions_dir, logger)
        self._definitions: Dict[str, WorkflowDefinition] = {}
        self._action_registry = ActionRegistry(logger)
        self._persistence = WorkflowPersistence(db_path, logger)

        # 实例管理
        self._instances: Dict[str, WorkflowInstance] = {}
        self._semaphore = asyncio.Semaphore(max_concurrent)

        # 回调
        self.on_instance_started: Optional[Callable] = None   # (instance)
        self.on_instance_completed: Optional[Callable] = None # (instance)
        self.on_instance_failed: Optional[Callable] = None    # (instance, error)

    # ═══════════════════════════════════════════════════════════
    # 初始化
    # ═══════════════════════════════════════════════════════════

    def load_definitions(self) -> int:
        """加载工作流定义"""
        self._definitions = self._def_loader.load_all()
        return len(self._definitions)

    async def recover_all(self) -> int:
        """恢复所有未完成的实例（断电恢复）"""
        rows = self._persistence.get_incomplete_instances()
        count = 0
        for row in rows:
            try:
                definition = self._definitions.get(row.get('workflow_name', ''))
                if not definition:
                    continue
                context = self._deserialize_context(row.get('context_blob'))
                instance = self._create_instance_obj(
                    row['task_id'], definition, context
                )
                instance.current_state = row.get('current_state', definition.initial_state)
                self._instances[instance.id] = instance
                asyncio.create_task(self._run_instance(instance))
                count += 1
                self._log_info(f"恢复实例: {instance.id} 状态={instance.current_state}")
            except Exception as e:
                self._log_error(f"恢复实例失败 {row.get('task_id')}: {e}")
        return count

    # ═══════════════════════════════════════════════════════════
    # 实例管理
    # ═══════════════════════════════════════════════════════════

    async def create_instance(self,
                              workflow_name: str,
                              context: Dict[str, Any]) -> Optional[str]:
        """创建并启动一个新工作流实例

        Args:
            workflow_name: 工作流名称（对应 YAML 文件名 stem）
            context:       任务上下文 {src_bay, dst_bay, robot_id, cargo_type}

        Returns:
            实例 ID（task_id），失败返回 None
        """
        definition = self._definitions.get(workflow_name)
        if not definition:
            self._log_error(f"工作流定义未找到: {workflow_name}")
            return None

        task_id = context.get('task_id', self._generate_task_id())
        if task_id in self._instances:
            self._log_warning(f"实例已存在: {task_id}")
            return None

        instance = self._create_instance_obj(task_id, definition, context)
        self._instances[task_id] = instance

        asyncio.create_task(self._run_instance(instance))
        self._log_info(f"创建实例: {task_id} ({workflow_name})")
        return task_id

    async def inject_event(self, task_id: str, event: str,
                           data: Dict[str, Any] = None):
        """向指定实例注入外部事件"""
        instance = self._instances.get(task_id)
        if instance:
            await instance.inject_event(event, data)

    async def cancel_instance(self, task_id: str):
        """取消指定实例"""
        instance = self._instances.get(task_id)
        if instance:
            await instance.inject_event('user_cancel', {})

    # ═══════════════════════════════════════════════════════════
    # Action 管理
    # ═══════════════════════════════════════════════════════════

    def register_action(self, name: str, func):
        """注册 Action 函数"""
        self._action_registry.register(name, func)

    def register_actions(self, actions: Dict[str, Callable]):
        """批量注册 Action 函数"""
        self._action_registry.register_batch(actions)

    # ═══════════════════════════════════════════════════════════
    # 查询
    # ═══════════════════════════════════════════════════════════

    def get_instance(self, task_id: str) -> Optional[WorkflowInstance]:
        return self._instances.get(task_id)

    def get_active_count(self) -> int:
        return len(self._instances)

    def get_all_instances_from_db(self, limit: int = 100) -> list:
        return self._persistence.get_all_instances(limit)

    # ═══════════════════════════════════════════════════════════
    # 内部
    # ═══════════════════════════════════════════════════════════

    async def _run_instance(self, instance: WorkflowInstance):
        """在协程中执行实例（带并发控制）"""
        async with self._semaphore:
            try:
                if self.on_instance_started:
                    self.on_instance_started(instance)
                await instance.run()
            except Exception as e:
                if self.on_instance_failed:
                    self.on_instance_failed(instance, e)
            finally:
                if self.on_instance_completed:
                    self.on_instance_completed(instance)
                self._instances.pop(instance.id, None)

    def _create_instance_obj(self, task_id: str, definition: WorkflowDefinition,
                             context: dict) -> WorkflowInstance:
        instance = WorkflowInstance(
            task_id, definition, context,
            self._action_registry, self._persistence, self._logger
        )
        instance.on_state_changed = lambda sid: self._persistence.update_state(task_id, sid)
        return instance

    @staticmethod
    def _generate_task_id() -> str:
        import time
        return f"TASK_{int(time.time() * 1000)}"

    @staticmethod
    def _deserialize_context(blob) -> dict:
        if blob is None:
            return {}
        import pickle
        try:
            return pickle.loads(blob)
        except Exception:
            return {}

    def _log_info(self, msg: str):

            self._logger.info(f"[WorkflowEngine] {msg}")

    def _log_warning(self, msg: str):
        if self._logger:
            self._logger.warning(f"[WorkflowEngine] {msg}")

    def _log_error(self, msg: str):
```

```python
#!/usr/bin/env python3
"""
工作流事件桥 — 连接 EventBus 与 WorkflowEngine

职责：
  - 订阅 EventBus 中与 RCS 回调相关的事件
  - 将事件路由到对应的工作流实例
  - 将工作流状态变更事件发布回 EventBus
"""

import asyncio
import logging
from typing import Dict, Any, Optional


class WorkflowEventBridge:
    """EventBus ↔ WorkflowEngine 双向事件桥"""

    def __init__(self, logger: logging.Logger = None):
        self._logger = logger or logging.getLogger(__name__)
        self._engine = None              # WorkflowEngine 实例
        self._event_bus = None           # EventBus 实例

    def connect(self, engine, event_bus):
        """连接 WorkflowEngine 和 EventBus

        Args:
            engine:    WorkflowEngine 实例
            event_bus: EventBus 实例
        """
        self._engine = engine
        self._event_bus = event_bus

        # 设置 Engine 回调 → 发布到 EventBus
        engine.on_instance_started = self._on_instance_started
        engine.on_instance_completed = self._on_instance_completed
        engine.on_instance_failed = self._on_instance_failed

        self._log_info("事件桥已连接")

    # ═══════════════════════════════════════════════════════════
    # Engine → EventBus（发布）
    # ═══════════════════════════════════════════════════════════

    def _on_instance_started(self, instance):
        self._publish('workflow.task_started', {
            'task_id': instance.id,
            'src_bay': instance.context.get('src_bay'),
            'dst_bay': instance.context.get('dst_bay'),
            'robot_id': instance.context.get('robot_id'),
            'cargo_type': instance.context.get('cargo_type'),
        })

    def _on_instance_completed(self, instance):
        self._publish('workflow.task_completed', {
            'task_id': instance.id,
            'duration': instance.duration,
            'current_state': instance.current_state,
            'src_bay': instance.context.get('src_bay'),
            'robot_id': instance.context.get('robot_id'),
        })

    def _on_instance_failed(self, instance, error):
        self._publish('workflow.task_failed', {
            'task_id': instance.id,
            'error': str(error) if error else 'unknown',
            'state': instance.current_state,
        })

    # ═══════════════════════════════════════════════════════════
    # EventBus → Engine（订阅后消费）
    # ═══════════════════════════════════════════════════════════

    def subscribe_rcs_events(self):
        """订阅 RCS 回调事件，自动路由到工作流实例

        应在 Engine 连接后调用，由 plugin 的 _activate_impl 触发。
        """
        if not self._event_bus or not self._engine:
            return

        self._event_bus.subscribe('rcs.agv_callback', self._on_rcs_agv_callback)
        self._event_bus.subscribe('rcs.warn_callback', self._on_rcs_warn_callback)
        self._log_info("已订阅 RCS 回调事件")

    def _on_rcs_agv_callback(self, data: Dict[str, Any]):
        """RCS agvCallback → 注入工作流实例"""
        task_code = data.get('task_code', '')
        method = data.get('method', '')
        if task_code and self._engine:
            event = 'callback_completed' if method == 'end' else 'callback_exception'
            asyncio.ensure_future(
                self._engine.inject_event(task_code, event, data)
            )

    def _on_rcs_warn_callback(self, data: Dict[str, Any]):
        """RCS warnCallback → 注入工作流实例"""
        task_code = data.get('task_code', '')
        if task_code and self._engine:
            asyncio.ensure_future(
                self._engine.inject_event(task_code, 'callback_exception', data)
            )

    # ═══════════════════════════════════════════════════════════
    # 内部
    # ═══════════════════════════════════════════════════════════

    def _publish(self, event_type: str, data: dict):
        if self._event_bus:
            try:
                self._event_bus.publish(event_type, data)
            except Exception as e:
                if self._logger:
                    self._logger.warning(f"发布事件失败 {event_type}: {e}")

    def _log_info(self, msg: str):
        if self._logger:
            self._logger.info(f"[EventBridge] {msg}")
```

```python
#!/usr/bin/env python3
"""
工作时间窗口检查器 — 判断当前时间是否在可调度时段内

支持的配置格式：
  1. 简化格式（从 params.yaml 读取）:
     "0 8-18 * * 1-5"  → 星期一到五 08:00-18:00

  2. JSON 列表格式（从 SQLite work_schedule 读取）:
     [
       {"day_of_week": "mon-fri", "start_time": "08:00", "end_time": "18:00"},
       {"day_of_week": "sat",     "start_time": "08:00", "end_time": "12:00"}
     ]
"""

import time
import logging
from typing import List, Optional
from datetime import datetime


class ScheduleChecker:
    """工作时间窗口检查器"""

    def __init__(self, schedules: List[dict] = None,
                 default_schedule: str = None,
                 logger: logging.Logger = None):
        """
        Args:
            schedules:        工作时段列表 [{day_of_week, start_time, end_time}]
            default_schedule: 默认简化格式 "0 8-18 * * 1-5"
            logger:           日志记录器
        """
        self._logger = logger or logging.getLogger(__name__)
        self._schedules = schedules or []
        self._default_schedule = default_schedule

    def set_schedules(self, schedules: List[dict]):
        """更新工作时段（用于配置热加载）"""
        self._schedules = schedules or []

    def is_in_working_window(self) -> bool:
        """判断当前时间是否在任何一个工作时间窗口内

        Returns:
            True: 在工作窗口内，允许调度
            False: 不在工作窗口内，不发起新任务
        """
        now = datetime.now()
        weekday = now.strftime('%a').lower()  # mon, tue, ...

        # 优先使用 schedules 列表
        for sched in self._schedules:
            if not sched.get('enabled', True):
                continue
            if self._match_day(weekday, sched.get('day_of_week', '')):
                if self._match_time(now, sched.get('start_time', ''),
                                    sched.get('end_time', '')):
                    return True

        # 若无 schedules，尝试解析 default_schedule
        if not self._schedules and self._default_schedule:
            return self._parse_simple_format(now, weekday, self._default_schedule)

        # 无任何配置时默认允许
        return True

    @staticmethod
    def _match_day(weekday: str, day_spec: str) -> bool:
        """匹配星期几"""
        if not day_spec or day_spec == '*':
            return True

        day_map = {'mon': 0, 'tue': 1, 'wed': 2, 'thu': 3,
                    'fri': 4, 'sat': 5, 'sun': 6}
        current_day = day_map.get(weekday, -1)

        if '-' in day_spec:
            parts = day_spec.split('-')
            if len(parts) == 2:
                start = day_map.get(parts[0].lower(), -1)
                end = day_map.get(parts[1].lower(), -1)
                return start <= current_day <= end

        return weekday == day_spec.lower()

    @staticmethod
    def _match_time(now: datetime, start: str, end: str) -> bool:
        """判断当前时间是否在 start-end 之间"""
        try:
            now_minutes = now.hour * 60 + now.minute
            start_minutes = int(start.split(':')[0]) * 60 + int(start.split(':')[1])
            end_minutes = int(end.split(':')[0]) * 60 + int(end.split(':')[1])
            return start_minutes <= now_minutes <= end_minutes
        except (ValueError, IndexError):
            return True  # 解析失败默认允许

    @staticmethod
    def _parse_simple_format(now: datetime, weekday: str, spec: str) -> bool:
        """解析简化格式 "0 8-18 * * 1-5" 或 "8-18 mon-fri"

        格式: "HH-HH DOW-DOW" 或 "HH:HH-HH:HH DOW-DOW"
        """
        try:
            parts = spec.strip().split()
            if len(parts) < 1:
                return True

            # 时间范围
            time_part = parts[0] if parts else '0-24'
            day_part = parts[1] if len(parts) > 1 else '*'

            # 解析时间
            if '-' in time_part:
                t_start, t_end = time_part.split('-', 1)
                if not ScheduleChecker._match_time(now, t_start, t_end):
                    return False

            # 解析星期
            if day_part != '*' and day_part != '':
                if not ScheduleChecker._match_day(weekday, day_part):
                    return False

            return True

        except Exception:
            return True
```

```python
#!/usr/bin/env python3
"""
资源锁定器 — 防并发抢占

设计要点：
  - 触发器扫描匹配成功后，立即锁定资源（叉车、起点仓位、终点仓位）
  - 工作流完成后通过 release() 释放锁
  - 线程安全（threading.Lock）
"""

import threading
import time
import logging
from typing import Set, Optional


class ResourceLocker:
    """资源锁定器 — 防止多个任务抢占同一资源"""

    def __init__(self, logger: logging.Logger = None):
        self._logger = logger or logging.getLogger(__name__)
        self._lock = threading.Lock()
        self._locked_robots: Set[int] = set()
        self._locked_src_bays: Set[str] = set()
        self._locked_dst_bays: Set[str] = set()

    # ── 锁定 ──

    def lock(self, robot_id: int, src_bay: str, dst_bay: str) -> bool:
        """锁定资源（原子操作）

        Returns:
            True: 锁定成功
            False: 任一资源已被锁定，未执行任何锁定
        """
        with self._lock:
            if robot_id in self._locked_robots:
                self._log_debug(f"叉车已锁定: {robot_id}")
                return False
            if src_bay in self._locked_src_bays:
                self._log_debug(f"起始仓位已锁定: {src_bay}")
                return False
            if dst_bay in self._locked_dst_bays:
                self._log_debug(f"终点仓位已锁定: {dst_bay}")
                return False

            self._locked_robots.add(robot_id)
            self._locked_src_bays.add(src_bay)
            self._locked_dst_bays.add(dst_bay)
            self._log_debug(f"锁定: robot={robot_id} src={src_bay} dst={dst_bay}")
            return True

    # ── 释放 ──

    def release(self, robot_id: int = None, src_bay: str = None,
                dst_bay: str = None):
        """释放资源"""
        with self._lock:
            if robot_id is not None:
                self._locked_robots.discard(robot_id)
            if src_bay is not None:
                self._locked_src_bays.discard(src_bay)
            if dst_bay is not None:
                self._locked_dst_bays.discard(dst_bay)
            self._log_debug(f"释放: robot={robot_id} src={src_bay} dst={dst_bay}")

    def release_all(self):
        """释放所有资源"""
        with self._lock:
            self._locked_robots.clear()
            self._locked_src_bays.clear()
            self._locked_dst_bays.clear()
            self._log_debug("释放全部资源")

    # ── 查询 ──

    def is_robot_locked(self, robot_id: int) -> bool:
        with self._lock:
            return robot_id in self._locked_robots

    def is_src_locked(self, bay_id: str) -> bool:
        with self._lock:
            return bay_id in self._locked_src_bays

    def is_dst_locked(self, bay_id: str) -> bool:
        with self._lock:
            return bay_id in self._locked_dst_bays

    def get_locked_count(self) -> dict:
        with self._lock:
            return {
                'robots': len(self._locked_robots),
                'src_bays': len(self._locked_src_bays),
                'dst_bays': len(self._locked_dst_bays),
            }

    def _log_debug(self, msg: str):
        if self._logger:
            self._logger.debug(f"[ResourceLocker] {msg}")
```

```python
#!/usr/bin/env python3
"""
触发器引擎 — 扫描多源缓存，匹配条件，生成任务上下文

触发条件（AND 逻辑）:
  1. 当前时间在工作时间窗口内
  2. 指定类型的起始仓位已绑定且未分配任务
  3. 指定类型的终点仓位为空
  4. 有空闲叉车
"""

import logging
from typing import Optional, Dict, Any
from .schedule_checker import ScheduleChecker
from .resource_locker import ResourceLocker


class TaskContext:
    """任务上下文 — 触发器生成的调度任务"""
    __slots__ = ('task_id', 'src_bay', 'dst_bay', 'robot_id', 'cargo_type')

    def __init__(self, task_id: str, src_bay: str, dst_bay: str,
                 robot_id: int, cargo_type: int):
        self.task_id = task_id
        self.src_bay = src_bay
        self.dst_bay = dst_bay
        self.robot_id = robot_id
        self.cargo_type = cargo_type

    def to_dict(self) -> Dict[str, Any]:
        return {
            'task_id': self.task_id,
            'src_bay': self.src_bay,
            'dst_bay': self.dst_bay,
            'robot_id': self.robot_id,
            'cargo_type': self.cargo_type,
        }


class TriggerEngine:
    """触发器引擎"""

    def __init__(self,
                 bay_cache,
                 dest_cache,
                 robot_cache,
                 config_cache,
                 logger: logging.Logger = None):
        """
        Args:
            bay_cache:    BayCache 实例
            dest_cache:   DestBayCache 实例
            robot_cache:  RobotCache 实例
            config_cache: ConfigCache 实例
            logger:       日志记录器
        """
        self._bay_cache = bay_cache
        self._dest_cache = dest_cache
        self._robot_cache = robot_cache
        self._config_cache = config_cache
        self._logger = logger or logging.getLogger(__name__)
        self._locker = ResourceLocker(logger)
        self._schedule_checker = ScheduleChecker(logger=logger)

    # ── 配置 ──

    def set_schedules(self, schedules: list):
        """更新工作时段"""
        self._schedule_checker.set_schedules(schedules)

    # ── 扫描 ──

    def scan(self, cargo_type: int) -> Optional[TaskContext]:
        """扫描指定货物类型，返回可执行的任务上下文

        Args:
            cargo_type: 货物类型 1~6

        Returns:
            TaskContext: 条件满足且资源锁定成功
            None:        条件不满足或资源已被占用
        """
        # 1. 检查时间窗口
        if not self._schedule_checker.is_in_working_window():
            self._log_debug(f"不在工作窗口内，跳过 cargo_type={cargo_type}")
            return None

        # 2. 查找已绑定且未分配任务的起始仓位
        src_bay = self._bay_cache.get_any_bound_bay(cargo_type)
        if not src_bay:
            return None

        # 3. 查找空的终点仓位
        dst_bay = self._dest_cache.get_any_empty_bay(cargo_type)
        if not dst_bay:
            return None

        # 4. 查找空闲叉车
        robot_id = self._robot_cache.get_idle_robot()
        if robot_id is None:
            return None

        # 5. 原子锁定
        if not self._locker.lock(robot_id, src_bay, dst_bay):
            return None

        # 6. 标记仓位在任务中
        self._bay_cache.mark_in_task(src_bay, True)

        # 7. 生成任务 ID 和上下文
        task_id = self._gen_task_id(cargo_type)
        ctx = TaskContext(task_id, src_bay, dst_bay, robot_id, cargo_type)

        self._log_info(f"生成任务: {task_id} src={src_bay} dst={dst_bay} "
                       f"robot={robot_id} type={cargo_type}")
        return ctx

    def scan_all_cargo_types(self) -> list:
        """扫描所有 6 种货物类型，返回所有可调度任务"""
        results = []
        for ct in range(1, 7):
            ctx = self.scan(ct)
            if ctx:
                results.append(ctx)
        return results

    # ── 释放 ──

    def release_task_resources(self, task_context: dict):
        """释放任务占用的资源"""
        self._locker.release(
            robot_id=task_context.get('robot_id'),
            src_bay=task_context.get('src_bay'),
        )
        src_bay = task_context.get('src_bay')
        if src_bay:
            self._bay_cache.mark_in_task(src_bay, False)

    def release_all(self):
        """释放所有锁定的资源"""
        self._locker.release_all()

    # ── 查询 ──

    def get_locked_count(self) -> dict:
        return self._locker.get_locked_count()

    # ── 内部 ──

    @staticmethod
    def _gen_task_id(cargo_type: int) -> str:
        import time
        return f"TASK_{int(time.time())}_{cargo_type}"

    def _log_info(self, msg: str):
        if self._logger:
            self._logger.info(f"[TriggerEngine] {msg}")

    def _log_debug(self, msg: str):
        if self._logger:
            self._logger.debug(f"[TriggerEngine] {msg}")
```
