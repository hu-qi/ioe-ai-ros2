## 3. 代码变更说明

### 3.1 阶段 1 核心修复（task_queue.py `remove_by_src`）

```python
def remove_by_src(self, src_bay: str):
    with self._lock:
        removed = [c for c in self._queue if c.src_bay == src_bay]
        self._queue = [c for c in self._queue if c.src_bay != src_bay]
        self._queued_src.discard(src_bay)
        # ★ 修复（doc/54 阶段1）：同步清除被移除候选的目的仓排除项，
        #   否则 _queued_dst 残留导致该目的仓无法再次匹配
        for c in removed:
            self._queued_dst.discard(c.dst_bay)
        if self._dest_cache is not None:
            for c in removed:
                self._dest_cache.release_occupied(
                    c.dst_bay, set_empty=True, source='unbind')
        # 防御（doc/54 N-2）：队列候选未加锁，release 幂等；
        # 已派发候选已出队不在 removed 中，不会误释放执行中任务
        if removed:
            self._locker.release(src_bay=src_bay)
```

### 3.2 阶段 3 防御（task_queue.py `_purge_unbound_locked` + `try_dequeue`）

```python
def try_dequeue(self):
    with self._lock:
        self._age_out_locked()
        # 防御（doc/54 N-1）：始发仓已解绑且 bay.unbound 事件丢失时，清理失效候选
        self._purge_unbound_locked()
        ...

def _purge_unbound_locked(self) -> int:
    """仅在缓存明确给出 bind_status 且非已绑定时判定失效（兼容无该字段的桩/旧缓存）。"""
    ...
    for c in self._queue:
        st = self._bay_cache.get_bay_status(c.src_bay)
        if (st is not None and st.get('bind_status') is not None
                and st.get('bind_status') != 1):
            removed += 1
            self._queued_src.discard(c.src_bay)
            self._queued_dst.discard(c.dst_bay)
            self._dest_cache.release_occupied(c.dst_bay, set_empty=True, source='unbind')
            self._skip_unbound_count += 1
        else:
            keep.append(c)
```
