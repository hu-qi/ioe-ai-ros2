## 2. Step 1: 新建告警分类器 alarm_classifier.py

**文件路径**: `app_mgr_object/components/workflow/alarm_classifier.py` **(新建)**

```python
#!/usr/bin/env python3
"""
告警分类器 — 根据 RCS warnCallback 数据判定告警等级和处置策略

参考: RCS-2000 V3.3 接口文档 §2.2.2 warnCallback

warnCallback 推送字段:
  warnCode:   告警编码 (如 "BATTERY_LOW", "COLLISION")
  warnMsg:    告警描述
  agvCode:    AGV 编号
  taskCode:   关联任务编号
  warnLevel:  告警等级 "0"(信息)/"1"(警告)/"2"(严重)/"3"(紧急)
"""

from enum import IntEnum
from dataclasses import dataclass
from typing import Dict, Any


class AlarmLevel(IntEnum):
    INFO     = 0   # 信息 — 仅记录, 不中断任务
    WARNING  = 1   # 警告 — 短暂暂停, 60s 自动恢复
    SEVERE   = 2   # 严重 — 转入人工介入, 持续监控
    CRITICAL = 3   # 紧急 — 转入人工介入 + 全局告警


@dataclass
class AlarmDisposition:
    """告警处置决策"""
    level: AlarmLevel
    engine_event: str          # 注入工作流的事件名
    should_pause: bool         # 是否暂停任务执行
    auto_recover_after: float  # 自动恢复等待秒数 (0=不自动恢复)
    description: str           # 可读描述


class AlarmClassifier:
    """告警分类器 — 按 warnCode 关键词匹配 + warnLevel 兜底"""

    # ═══════════════════════════════════════════════════════
    # 预设规则: {warnCode 关键词 → AlarmDisposition}
    # 匹配方式: warn_data['warn_code'] 包含关键词即命中
    # ═══════════════════════════════════════════════════════
    RULES: Dict[str, AlarmDisposition] = {
        # L0 信息 — 不影响任务
        'BATTERY_LOW':   AlarmDisposition(AlarmLevel.INFO,    'warn_info',    False, 0,   '电池电量低'),
        'APPROACHING':   AlarmDisposition(AlarmLevel.INFO,    'warn_info',    False, 0,   '即将到达目标'),
        'SPEED_REDUCE':  AlarmDisposition(AlarmLevel.INFO,    'warn_info',    False, 0,   '速度降低'),

        # L1 警告 — 短暂暂停后自动恢复
        'PATH_BLOCKED':  AlarmDisposition(AlarmLevel.WARNING, 'warn_l1',      True,  60,  '路径阻塞(可绕行)'),
        'SIGNAL_WEAK':   AlarmDisposition(AlarmLevel.WARNING, 'warn_l1',      True,  30,  '信号弱'),
        'OBSTACLE':      AlarmDisposition(AlarmLevel.WARNING, 'warn_l1',      True,  60,  '障碍物检测'),
        'TRAFFIC_JAM':   AlarmDisposition(AlarmLevel.WARNING, 'warn_l1',      True,  60,  '交通拥堵'),

        # L2 严重 — 需人工介入
        'COLLISION':     AlarmDisposition(AlarmLevel.SEVERE,  'callback_exception', True, 0, 'AGV 碰撞'),
        'MECH_FAILURE':  AlarmDisposition(AlarmLevel.SEVERE,  'callback_exception', True, 0, '机械故障'),
        'MOTOR_ERROR':   AlarmDisposition(AlarmLevel.SEVERE,  'callback_exception', True, 0, '电机异常'),
        'DERAIL':        AlarmDisposition(AlarmLevel.SEVERE,  'callback_exception', True, 0, '脱轨'),
        'LIFT_FAILURE':  AlarmDisposition(AlarmLevel.SEVERE,  'callback_exception', True, 0, '举升故障'),

        # L3 紧急 — 全局告警
        'EMERGENCY':     AlarmDisposition(AlarmLevel.CRITICAL,'callback_exception', True, 0, '紧急停止'),
        'FIRE':          AlarmDisposition(AlarmLevel.CRITICAL,'callback_exception', True, 0, '火灾告警'),
        'POWER_OFF':     AlarmDisposition(AlarmLevel.CRITICAL,'callback_exception', True, 0, '断电'),
    }

    # ═══════════════════════════════════════════════════════
    # 兜底规则: 按 warnLevel 数值
    # ═══════════════════════════════════════════════════════
    LEVEL_DEFAULTS: Dict[int, AlarmDisposition] = {
        0: AlarmDisposition(AlarmLevel.INFO,    'warn_info',           False, 0,   '未知信息告警'),
        1: AlarmDisposition(AlarmLevel.WARNING, 'warn_l1',             True,  60,  '未知警告告警'),
        2: AlarmDisposition(AlarmLevel.SEVERE,  'callback_exception',  True,  0,   '未知严重告警'),
        3: AlarmDisposition(AlarmLevel.CRITICAL,'callback_exception',  True,  0,   '未知紧急告警'),
    }

    # ═══════════════════════════════════════════════════════
    #  分类方法
    # ═══════════════════════════════════════════════════════

    @classmethod
    def classify(cls, warn_data: Dict[str, Any]) -> AlarmDisposition:
        """根据告警数据判定处置策略

        优先按 warnCode 关键词匹配预设规则，未匹配则按 warnLevel 兜底。

        Args:
            warn_data: {
                'warn_code':  str,    # 告警编码
                'warn_msg':   str,    # 告警描述
                'warn_level': str,    # "0"/"1"/"2"/"3"
                'agv_code':   str,
                'task_code':  str,
            }

        Returns:
            AlarmDisposition 处置决策
        """
        warn_code = warn_data.get('warn_code', '').upper()

        # 1. 按 warnCode 关键词匹配
        for keyword, disposition in cls.RULES.items():
            if keyword in warn_code:
                return disposition

        # 2. 按 warnLevel 兜底
        try:
            level = int(warn_data.get('warn_level', '0'))
        except (ValueError, TypeError):
            level = 0

        return cls.LEVEL_DEFAULTS.get(
            level,
            cls.LEVEL_DEFAULTS[0]  # 终极兜底: 按 INFO 处理
        )
```

### 3.2 替换 `_on_rcs_warn_callback` 方法（约第 100 行）

**定位**: 找到 `def _on_rcs_warn_callback(self, data: Dict[str, Any]):` 方法并完整替换为:

```python
def _on_rcs_warn_callback(self, data: Dict[str, Any]):
    """RCS warnCallback → 告警分级路由处置"""
    task_code = data.get('task_code', '')
    if not task_code or not self._engine:
        return

    # 检查任务实例是否存在
    instance = self._engine.get_instance(task_code)
    if not instance:
        self._logger.warning(f"收到 warnCallback 但未找到对应实例: {task_code}")
        return

    # === 告警分级判定 ===
    disposition = AlarmClassifier.classify(data)

    self._logger.info(
        f"RCS warnCallback 分级: [{disposition.level.name}] {disposition.description} "
        f"→ event={disposition.engine_event} "
        f"(warnCode={data.get('warn_code', '')})"
    )

    # === 发布告警日志事件 (供 persistence 层记录) ===
    self._publish('rcs.alarm_logged', {
        'task_code': task_code,
        'alarm_level': disposition.level.name,
        'warn_code': data.get('warn_code', ''),
        'warn_msg': data.get('warn_msg', ''),
        'agv_code': data.get('agv_code', ''),
        'disposition': disposition.engine_event,
    })

    # === 按等级执行不同策略 ===

    if disposition.level == AlarmLevel.INFO:
        # L0: 仅记录，不注入任何事件到工作流实例
        return

    elif disposition.level == AlarmLevel.WARNING:
        # L1: 注入 warn_l1 → 工作流进入 warn_pause 状态
        self._inject_to_engine(task_code, disposition.engine_event, data)
        return

    elif disposition.level in (AlarmLevel.SEVERE, AlarmLevel.CRITICAL):
        # L2/L3: 注入 callback_exception → manual_intervention
        self._inject_to_engine(task_code, disposition.engine_event, data)

        # L3 额外发布全局紧急告警 (供 Web 面板弹出)
        if disposition.level == AlarmLevel.CRITICAL:
            self._publish('rcs.critical_alarm', {
                'task_code': task_code,
                'warn_msg': data.get('warn_msg', ''),
                'agv_code': data.get('agv_code', ''),
                'level': 'CRITICAL',
            })


def _inject_to_engine(self, task_code: str, event: str, data: dict):
    """安全注入事件到工作流引擎 (使用正确的事件循环)"""
    import asyncio
    try:
        if not hasattr(self, '_loop') or self._loop is None:
            self._logger.error(f"事件循环未初始化，无法注入事件: {task_code}")
            return

        future = asyncio.run_coroutine_threadsafe(
            self._engine.inject_event(task_code, event, data),
            self._loop
        )
        future.add_done_callback(
            lambda f: self._logger.debug(f"事件注入完成: {task_code}")
            if f.exception() is None
            else self._logger.error(f"事件注入失败: {task_code} {f.exception()}")
        )
    except Exception as e:
        self._logger.error(f"注入事件异常: {task_code} → {event}: {e}")
```

> **注意**: `_inject_to_engine` 放在 `WorkflowEventBridge` 类内部（与 `_on_rcs_warn_callback` 同级缩进），替代原来 `_on_rcs_warn_callback` 中的 `asyncio.ensure_future` 调用。

```python
def get_all_active_monitorable_instances(self) -> Dict[str, str]:
    """获取所有需要轮询监控的活跃实例

    覆盖 monitoring + manual_intervention 两种状态。
    task_status_monitor 插件用此方法获取批量轮询对象。

    Returns:
        {task_code: rcs_task_no}  映射
    """
    result = {}
    for task_id, instance in self._instances.items():
        if not instance.is_running:
            continue
        if instance.current_state not in ('monitoring', 'manual_intervention'):
            continue
        rcs_task_no = instance.context.get('rcs_task_no', '')
        if rcs_task_no:
            result[task_id] = rcs_task_no
    return result
```

```python
def log_alarm(self, task_id: str, alarm_level: str,
              warn_code: str = '', warn_msg: str = '',
              agv_code: str = '', disposition: str = ''):
    """记录告警日志 (非阻塞, 失败不影响主流程)"""
    try:
        with sqlite3.connect(self._db_path) as conn:
            conn.execute("""
                INSERT INTO alarm_logs
                (task_id, alarm_level, warn_code, warn_msg, agv_code, disposition)
                VALUES (?, ?, ?, ?, ?, ?)
            """, (task_id, alarm_level, warn_code, warn_msg, agv_code, disposition))
            conn.commit()
    except Exception:
        pass

def log_state_change(self, task_id: str, from_state: str,

    except Exception:
        pass

def log_exception(self, task_id: str, event_type: str, detail: str = ''):

    try:
        with sqlite3.connect(self._db_path) as conn:

                INSERT INTO exception_log (task_id, event_type, detail)
                VALUES (?, ?, ?)
            """, (task_id, event_type, detail))
            conn.commit()
    except Exception:
        pass

def query_alarms_by_task(self, task_id: str, limit: int = 50) -> list:
    """按任务 ID 查询告警日志"""
    try:
        with sqlite3.connect(self._db_path) as conn:
            conn.row_factory = sqlite3.Row
```

```python
async def _action_log_manual_entry(self, context: dict, instance) -> str:
    """记录进入 manual_intervention 的时间戳 + 触发原因"""
    import time
    context['manual_entered_at'] = time.time()
    context['manual_trigger_event'] = context.get('_last_event', 'unknown')

    self.logger.warning(
        f"任务进入人工介入: {context.get('task_id')} "
        f"触发事件={context['manual_trigger_event']}, "
        f"时间戳={context['manual_entered_at']}"
    )

    try:
        self._engine._persistence.log_exception(
            context.get('task_id', ''),
            'manual_intervention',
            f"trigger={context['manual_trigger_event']}"
        )
    except Exception:
        pass

    return 'done'
```

```python
def _poll_once(self):
    """定时器回调: 执行一轮按需轮询 (monitoring + manual_intervention)"""
    try:
        engine = self._get_workflow_engine()
        if not engine:
            return

        # === 1. 监控 monitoring 实例 (原有逻辑) ===
        monitoring = engine.get_active_monitoring_instances()
        if monitoring:
            rcs = self._get_rcs_adapter()
            if rcs:
                self._monitor.poll_if_needed(rcs, monitoring)

        # === 2. 新增: 监控 manual_intervention 实例 (低频率) ===
        if self._should_poll_manual():
            all_monitorable = engine.get_all_active_monitorable_instances()
            manual_only = {
                k: v for k, v in all_monitorable.items()
                if k not in monitoring
            }
            if manual_only:
                rcs = self._get_rcs_adapter()
                if rcs:
                    self._poll_manual_instances(rcs, manual_only, engine)

        # === 3. 判断是否停止 timer ===
        if not monitoring and not engine.get_all_active_monitorable_instances():
            if self._monitor.should_stop_timer():
                self._stop_poll_timer()
                return

        self._total_polls += 1

    except Exception as e:
        self.logger.warning(f"轮询执行异常: {e}")
```

### 8.4 在类中新增方法（放在 `_poll_once` 之后）

```python
# ====== 新增: manual_intervention 轮询方法 ======

def _should_poll_manual(self) -> bool:
    """manual_intervention 实例轮询频率控制 (每 30s 一次)"""
    now = time.time()
    if now - self._last_manual_poll >= self._manual_poll_interval:
        self._last_manual_poll = now
        return True
    return False


def _poll_manual_instances(self, rcs, manual_instances: dict, engine):
    """对 manual_intervention 实例查询 RCS 状态

    检测两种自动闭环场景:
      - RCS 返回 COMPLETED(9) → task_monitor 注入 auto_resolved
      - RCS 返回 FAILED(7)/CANCELLED(8) → task_monitor 注入 auto_failed

    同时在超时未处理时触发告警升级。
    """
    import asyncio
    import json

    task_codes = list(manual_instances.keys())
    self.logger.debug(f"manual_intervention 轮询: {len(task_codes)} 个任务")

    try:
        resp = rcs.query_task_status(task_codes)
        if resp.get('code') != '0':
            return

        now = time.time()
        for item in resp.get('data', []):
            task_code = item.get('taskCode', '')
            task_status = str(item.get('taskStatus', ''))

            if task_code not in engine._instances:
                continue

            instance = engine._instances[task_code]
            if instance.current_state != 'manual_intervention':
                continue

            # === 自动闭环检测 ===
            if task_status == '9':
                self.logger.info(
                    f"自动闭环: {task_code} RCS 已完成(9), 注入 auto_resolved"
                )
                future = asyncio.run_coroutine_threadsafe(
                    engine.inject_event(task_code, 'auto_resolved', {
                        'task_status': task_status,
                        'source': 'task_monitor_auto_closure',
                    }),
                    self.node.asyncio_loop
                )
                future.add_done_callback(
                    lambda f, tc=task_code: self.logger.debug(
                        f"auto_resolved 注入完成: {tc}"
                    ) if f.exception() is None
                    else self.logger.error(f"auto_resolved 注入失败: {tc}")
                )

            elif task_status in ('7', '8'):
                self.logger.info(
                    f"自动闭环: {task_code} RCS 已终结({task_status}), 注入 auto_failed"
                )
                future = asyncio.run_coroutine_threadsafe(
                    engine.inject_event(task_code, 'auto_failed', {
                        'task_status': task_status,
                        'source': 'task_monitor_auto_closure',
                    }),
                    self.node.asyncio_loop
                )
                future.add_done_callback(
                    lambda f, tc=task_code: self.logger.debug(
                        f"auto_failed 注入完成: {tc}"
                    ) if f.exception() is None
                    else self.logger.error(f"auto_failed 注入失败: {tc}")
                )

            # === 超时升级检测 ===
            entered_at = instance.context.get('manual_entered_at', 0)
            if entered_at and (now - entered_at) > self._manual_timeout:
                self.logger.warning(
                    f"告警升级: {task_code} 在 manual_intervention 已超 "
                    f"{int(now - entered_at)}s, 等待人工处置"
                )
                self._publish_event('rcs.alarm_escalated', {
                    'task_id': task_code,
                    'duration': now - entered_at,
                    'agv_code': instance.context.get('robot_id', ''),
                })

    except Exception as e:
        self.logger.warning(f"manual_intervention 轮询异常: {e}")

# ==================================================
```
